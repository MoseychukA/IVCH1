//+++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++

Ниже — рабочий “скелет” WEB‑интерфейса для STM32F103 + W5500 (Ethernet_Generic),который:

показывает текущие настройки устройства (INTERNET1/INTERNET2/GSM + статусы),
позволяет из браузера изменить параметры INTERNET1 и INTERNET2 (DHCP/IP/MASK/GW/DNS + NTP IP idx + период),
позволяет включить/выключить и настроить GSM (минимально:Enable/оператор/период; APN/логин/пароль можно добавить следующим этапом),
после сохранения вызывает ваши функции:
applyInternet1FromStore();
applyInternet2FromStore();
(для GSM) planner.onSettingsChanged(); или вашу функцию применения.
Код написан максимально “по‑Arduino”:EthernetServer + простые HTML формы,без JS.

Важно:я использую ваши уже существующие хранилища lanStore (IF1/IF2) и syncStore/syncData (GSM). 
Если у вас имена/структуры отличаются — скажите,я подгоню.

bool WebUI::startsWith(const char* s,const char* pref)
{
 while (*pref) {
 if (*s++ != *pref++) return false;
 }
 return true;
}

bool WebUI::readRequestLine(EthernetClient& c,char* out,size_t outN,uint32_t timeoutMs)
{
 uint32_t t0 = millis();
 size_t n = 0;

 while ((millis() - t0) < timeoutMs) {
 while (c.available()) {
 char ch = (char)c.read();
 if (ch == '\r') continue;
 if (ch == '\n') {
 out[n] = 0;
 return (n > 0);
 }
 if (n + 1 < outN) out[n++] = ch;
 }
 }
 out[0] = 0;
 return false;
}

void WebUI::drainHeaders(EthernetClient& c,uint32_t timeoutMs)
{
 uint32_t t0 = millis();
 int crlfCount = 0;
 while ((millis() - t0) < timeoutMs) {
 while (c.available()) {
 char ch = (char)c.read();
 // ждём пустую строку \r\n\r\n (мы уже игнорим \r при чтении line,но здесь проще по \n\n)
 if (ch == '\n') {
 crlfCount++;
 if (crlfCount >= 2) return;
 } else if (ch != '\r') {
 crlfCount = 0;
 }
 }
 }
}

void WebUI::urlDecodeInPlace(char* s)
{
 auto hex = [](char c)->int {
 if (c >= '0' && c <= '9') return c - '0';
 if (c >= 'A' && c <= 'F') return 10 + (c - 'A');
 if (c >= 'a' && c <= 'f') return 10 + (c - 'a');
 return -1;
 };

 char* w = s;
 for (char* r = s; *r; r++) {
 if (*r == '+') { *w++ = ' '; continue; }
 if (*r == '%' && r[1] && r[2]) {
 int hi = hex(r[1]),lo = hex(r[2]);
 if (hi >= 0 && lo >= 0) {
 *w++ = (char)((hi << 4) | lo);
 r += 2;
 continue;
 }
 }
 *w++ = *r;
 }
 *w = 0;
}

const char* WebUI::findParam(char* query,const char* key)
{
 // query:"a=1&b=2"
 // вернём указатель на value внутри query или nullptr
 size_t klen = strlen(key);
 char* p = query;

 while (p && *p) {
 char* amp = strchr(p,'&');
 if (amp) *amp = 0;

 char* eq = strchr(p,'=');
 if (eq) {
 *eq = 0;
 if (strlen(p) == klen && strcmp(p,key) == 0) {
 char* val = eq + 1;
 if (amp) *amp = '&';
 *eq = '=';
 return val;
 }
 *eq = '=';
 }

 if (amp) {
 *amp = '&';
 p = amp + 1;
 } else break;
 }
 return nullptr;
}

bool WebUI::parseBool01(const char* s,bool& out)
{
 if (!s) return false;
 if (strcmp(s,"0") == 0) { out = false; return true; }
 if (strcmp(s,"1") == 0) { out = true; return true; }
 return false;
}

bool WebUI::parseU8(const char* s,uint8_t& out,uint8_t minV,uint8_t maxV)
{
 if (!s) return false;
 long v = strtol(s,nullptr,10);
 if (v < minV || v > maxV) return false;
 out = (uint8_t)v;
 return true;
}

bool WebUI::parseIp4(const char* s,uint8_t out[4])
{
 if (!s) return false;
 int a,b,c,d;
 if (sscanf(s,"%d.%d.%d.%d",&a,&b,&c,&d) != 4) return false;
 if ((unsigned)a>255 || (unsigned)b>255 || (unsigned)c>255 || (unsigned)d>255) return false;
 out[0]=(uint8_t)a; out[1]=(uint8_t)b; out[2]=(uint8_t)c; out[3]=(uint8_t)d;
 return true;
}

void WebUI::sendHeader(EthernetClient& c,int code,const __FlashStringHelper* ct)
{
 if (code == 200) c.print(F("HTTP/1.1 200 OK\r\n"));
 else if (code == 303) c.print(F("HTTP/1.1 303 See Other\r\n"));
 else c.print(F("HTTP/1.1 404 Not Found\r\n"));

 c.print(F("Connection:close\r\n"));
 c.print(F("Cache-Control:no-store\r\n"));
 c.print(F("Content-Type:"));
 c.print(ct);
 c.print(F("\r\n\r\n"));
}

void WebUI::sendRedirect(EthernetClient& c,const char* location)
{
 c.print(F("HTTP/1.1 303 See Other\r\n"));
 c.print(F("Connection:close\r\n"));
 c.print(F("Location:"));
 c.print(location);
 c.print(F("\r\n\r\n"));
}

void WebUI::htmlEscPrint(EthernetClient& c,const char* s)
{
 for (; *s; s++) {
 switch (*s) {
 case '&':c.print(F("&amp;")); break;
 case '<':c.print(F("&lt;")); break;
 case '>':c.print(F("&gt;")); break;
 case '"':c.print(F("&quot;")); break;
 default:c.write(*s); break;
 }
 }
}

void WebUI::handleClient(EthernetClient& c)
{
 char line[192];
 if (!readRequestLine(c,line,sizeof(line),1500)) { c.stop(); return; }

 // Пример:
 // GET /internet1 HTTP/1.1
 // POST /internet1/save HTTP/1.1
 drainHeaders(c,1500);

 bool isGet = startsWith(line,"GET ");
 bool isPost = startsWith(line,"POST ");
 if (!isGet && !isPost) { sendHeader(c,404,F("text/plain")); c.print(F("Bad method")); c.stop(); return; }

 // вытащим path (до пробела)
 const char* p = isGet ? (line + 4) :(line + 5);
 char path[128];
 size_t i = 0;
 while (*p && *p != ' ' && i + 1 < sizeof(path)) path[i++] = *p++;
 path[i] = 0;

 // query (для простоты:в POST будем принимать параметры через URL query,т.е. /internet1/save?....)
 char* q = strchr(path,'?');
 char* query = nullptr;
 if (q) { *q = 0; query = q + 1; urlDecodeInPlace(query); }

 // routes
 if (isGet && strcmp(path,"/") == 0) { pageHome(c); }
 else if (isGet && strcmp(path,"/internet1") == 0) { pageInternet1(c); }
 else if (isGet && strcmp(path,"/internet2") == 0) { pageInternet2(c); }
 else if (isGet && strcmp(path,"/gsm") == 0) { pageGsm(c); }
 else if (isPost && strcmp(path,"/internet1/save") == 0) { actionInternet1Save(c,query); }
 else if (isPost && strcmp(path,"/internet2/save") == 0) { actionInternet2Save(c,query); }
 else if (isPost && strcmp(path,"/gsm/save") == 0) { actionGsmSave(c,query); }
 else { pageNotFound(c); }

 c.stop();
}

// ---------------- Actions ----------------
// Параметры принимаем из query в URL (быстро и просто).
// Пример:POST /internet1/save?dhcp=1&ntpIdx=0&perIdx=3&ip=... HTTP/1.1

void WebUI::actionInternet1Save(EthernetClient& c,char* query)
{
 if (!query) { sendHeader(c,404,F("text/plain")); c.print(F("No query")); return; }

 LanIfStore::IfConfig cfg{};
 lanStore.load(LanIfStore::IF1,cfg,false);

 // parse
 bool dhcp;
 uint8_t ntpIdx,perIdx;
 uint8_t ip[4],mask[4],gw[4],dns[4];

 if (parseBool01(findParam(query,"dhcp"),dhcp)) cfg.dhcp = dhcp ? 1 :0;
 if (parseU8(findParam(query,"ntpIdx"),ntpIdx,0,4)) cfg.ntpIdx = ntpIdx;
 if (parseU8(findParam(query,"perIdx"),perIdx,0,5)) cfg.periodIdx = perIdx;

 const char* sip = findParam(query,"ip");
 const char* smask = findParam(query,"mask");
 const char* sgw = findParam(query,"gw");
 const char* sdns = findParam(query,"dns");

 if (sip && parseIp4(sip,ip)) memcpy(cfg.ip,ip,4);
 if (smask && parseIp4(smask,mask)) memcpy(cfg.mask,mask,4);
 if (sgw && parseIp4(sgw,gw)) memcpy(cfg.gw,gw,4);
 if (sdns && parseIp4(sdns,dns)) memcpy(cfg.dns,dns,4);

 lanStore.save(LanIfStore::IF1,cfg,true);
 applyInternet1FromStore();

 sendRedirect(c,"/internet1");
}

 if (parseU8(findParam(query,"ntpIdx"),ntpIdx,0,4)) cfg.ntpIdx = ntpIdx;
 if (parseU8(findParam(query,"perIdx"),perIdx,0,5)) cfg.periodIdx = perIdx;

 const char* sip = findParam(query,"ip");
 const char* smask = findParam(query,"mask");
 const char* sgw = findParam(query,"gw");
 const char* sdns = findParam(query,"dns");

 if (sip && parseIp4(sip,ip)) memcpy(cfg.ip,ip,4);

void WebUI::htmlEscPrint(EthernetClient& c,const char* s)
{
	for (; *s; s++) {
		switch (*s) {
		case '&':c.print(F("&amp;")); break;
		case '<':c.print(F("&lt;")); break;
		case '>':c.print(F("&gt;")); break;
		case '"':c.print(F("&quot;")); break;
		default:c.write(*s); break;
		}
	}
}

bool WebUI::startsWith(const char* s,const char* pref)

bool WebUI::readRequestLine(EthernetClient& c,char* out,size_t outN,uint32_t timeoutMs)
{
	uint32_t t0 = millis();
	size_t n = 0;

	while ((millis() - t0) < timeoutMs) {
		while (c.available()) {
			char ch = (char)c.read();
			if (ch == '\r') continue;
			if (ch == '\n') {
				out[n] = 0;
				return (n > 0);
			}
			if (n + 1 < outN) out[n++] = ch;
		}
	}
	out[0] = 0;
	return false;
}

void WebUI::drainHeaders(EthernetClient& c,uint32_t timeoutMs)
{
	uint32_t t0 = millis();
	int crlfCount = 0;
	while ((millis() - t0) < timeoutMs) {
		while (c.available()) {
			char ch = (char)c.read();
			if (ch == '\n') {
				crlfCount++;
				if (crlfCount >= 2) return;
			}
			else if (ch != '\r') {
				crlfCount = 0;
			}
		}
	}
}

void WebUI::urlDecodeInPlace(char* s)
{
	auto hex = [](char c)->int {
		if (c >= '0' && c <= '9') return c - '0';
		if (c >= 'A' && c <= 'F') return 10 + (c - 'A');
		if (c >= 'a' && c <= 'f') return 10 + (c - 'a');
		return -1;
	};

	char* w = s;
	for (char* r = s; *r; r++) {
		if (*r == '+') { *w++ = ' '; continue; }
		if (*r == '%' && r[1] && r[2]) {
			int hi = hex(r[1]),lo = hex(r[2]);
			if (hi >= 0 && lo >= 0) {
				*w++ = (char)((hi << 4) | lo);
				r += 2;
				continue;
			}
		}
		*w++ = *r;
	}
	*w = 0;
}

const char* WebUI::findParam(char* query,const char* key)
{
	size_t klen = strlen(key);
	char* p = query;

	while (p && *p) {
		char* amp = strchr(p,'&');
		if (amp) *amp = 0;

		char* eq = strchr(p,'=');
		if (eq) {
			*eq = 0;
			if (strlen(p) == klen && strcmp(p,key) == 0) {
				char* val = eq + 1;
				if (amp) *amp = '&';
				*eq = '=';
				return val;
			}
			*eq = '=';
		}

		if (amp) {
			*amp = '&';
			p = amp + 1;
		}
		else break;
	}
	return nullptr;
}

bool WebUI::parseBool01(const char* s,bool& out)
{
	if (!s) return false;
	if (strcmp(s,"0") == 0) { out = false; return true; }
	if (strcmp(s,"1") == 0) { out = true; return true; }
	return false;
}

bool WebUI::parseU8(const char* s,uint8_t& out,uint8_t minV,uint8_t maxV)
{
	if (!s) return false;
	long v = strtol(s,nullptr,10);
	if (v < minV || v > maxV) return false;
	out = (uint8_t)v;
	return true;
}

bool WebUI::parseIp4(const char* s,uint8_t out[4])
{
	if (!s) return false;
	int a,b,c,d;
	if (sscanf(s,"%d.%d.%d.%d",&a,&b,&c,&d) != 4) return false;
	if ((unsigned)a>255 || (unsigned)b>255 || (unsigned)c>255 || (unsigned)d>255) return false;
	out[0]=(uint8_t)a; out[1]=(uint8_t)b; out[2]=(uint8_t)c; out[3]=(uint8_t)d;
	return true;
}

void WebUI::sendHeader(EthernetClient& c,int code,const __FlashStringHelper* ct)
{
	if (code == 200) c.print(F("HTTP/1.1 200 OK\r\n"));
	else if (code == 303) c.print(F("HTTP/1.1 303 See Other\r\n"));
	else c.print(F("HTTP/1.1 404 Not Found\r\n"));

	c.print(F("Connection:close\r\n"));
	c.print(F("Cache-Control:no-store\r\n"));
	c.print(F("Content-Type:"));
	c.print(ct);
	c.print(F("\r\n\r\n"));
}

}

void WebUI::handleClient(EthernetClient& c)
{
	char line[192];
	if (!readRequestLine(c,line,sizeof(line),1500)) { c.stop(); return; }

	drainHeaders(c,1500);

	bool isGet = startsWith(line,"GET ");
	bool isPost = startsWith(line,"POST ");
	if (!isGet && !isPost) { sendHeader(c,404,F("text/plain")); c.print(F("Bad method")); c.stop(); return; }

	const char* p = isGet ? (line + 4) :(line + 5);
	char path[128];
	size_t i = 0;
	while (*p && *p != ' ' && i + 1 < sizeof(path)) path[i++] = *p++;
	path[i] = 0;

	char* q = strchr(path,'?');
	char* query = nullptr;
	if (q) { *q = 0; query = q + 1; urlDecodeInPlace(query); }

	if (isGet && strcmp(path,"/") == 0) { pageHome(c); }
	else if (isGet && strcmp(path,"/internet1") == 0) { pageInternet1(c); }
	else if (isGet && strcmp(path,"/internet2") == 0) { pageInternet2(c); }
	else if (isGet && strcmp(path,"/gsm") == 0) { pageGsm(c); }
	else if (isPost && strcmp(path,"/internet1/save") == 0) { actionInternet1Save(c,query); }
	else if (isPost && strcmp(path,"/internet2/save") == 0) { actionInternet2Save(c,query); }
	else if (isPost && strcmp(path,"/gsm/save") == 0) { actionGsmSave(c,query); }
	else { pageNotFound(c); }

	c.stop();
}

void WebUI::htmlEscPrint(EthernetClient& c,const char* s)
{
	for (; *s; s++) {
		switch (*s) {
		case '&':c.print(F("&amp;")); break;
		case '<':c.print(F("&lt;")); break;
		case '>':c.print(F("&gt;")); break;
		case '"':c.print(F("&quot;")); break;
		default:c.write(*s); break;
		}
	}
}

bool WebUI::readRequestLine(EthernetClient& c,char* out,size_t outN,uint32_t timeoutMs)
{
	uint32_t t0 = millis();
	size_t n = 0;

	while ((millis() - t0) < timeoutMs) {
		while (c.available()) {
			char ch = (char)c.read();
			if (ch == '\r') continue;
			if (ch == '\n') {
				out[n] = 0;
				return (n > 0);
			}
			if (n + 1 < outN) out[n++] = ch;
		}
	}
	out[0] = 0;
	return false;
}

void WebUI::drainHeaders(EthernetClient& c,uint32_t timeoutMs)
{
	uint32_t t0 = millis();
	int crlfCount = 0;
	while ((millis() - t0) < timeoutMs) {
		while (c.available()) {
			char ch = (char)c.read();
			if (ch == '\n') {
				crlfCount++;
				if (crlfCount >= 2) return;
			}
			else if (ch != '\r') {
				crlfCount = 0;
			}
		}
	}
}

void WebUI::urlDecodeInPlace(char* s)
{
	auto hex = [](char c)->int {
		if (c >= '0' && c <= '9') return c - '0';
		if (c >= 'A' && c <= 'F') return 10 + (c - 'A');
		if (c >= 'a' && c <= 'f') return 10 + (c - 'a');
		return -1;
	};

	char* w = s;
	for (char* r = s; *r; r++) {
		if (*r == '+') { *w++ = ' '; continue; }
		if (*r == '%' && r[1] && r[2]) {
			int hi = hex(r[1]),lo = hex(r[2]);
			if (hi >= 0 && lo >= 0) {
				*w++ = (char)((hi << 4) | lo);
				r += 2;
				continue;
			}
		}
		*w++ = *r;
	}
	*w = 0;
}

const char* WebUI::findParam(char* query,const char* key)
{
	size_t klen = strlen(key);
	char* p = query;

	while (p && *p) {
		char* amp = strchr(p,'&');
		if (amp) *amp = 0;

		char* eq = strchr(p,'=');
		if (eq) {
			*eq = 0;
			if (strlen(p) == klen && strcmp(p,key) == 0) {
				char* val = eq + 1;
				if (amp) *amp = '&';
				*eq = '=';
				return val;
			}
			*eq = '=';
		}

		if (amp) {
			*amp = '&';
			p = amp + 1;
		}
		else break;
	}
	return nullptr;
}

bool WebUI::parseBool01(const char* s,bool& out)
{
	if (!s) return false;
	if (strcmp(s,"0") == 0) { out = false; return true; }
	if (strcmp(s,"1") == 0) { out = true; return true; }
	return false;
}

bool WebUI::parseU8(const char* s,uint8_t& out,uint8_t minV,uint8_t maxV)
{
	if (!s) return false;
	long v = strtol(s,nullptr,10);
	if (v < minV || v > maxV) return false;
	out = (uint8_t)v;
	return true;
}

bool WebUI::parseIp4(const char* s,uint8_t out[4])
{
	if (!s) return false;
	int a,b,c,d;
	if (sscanf(s,"%d.%d.%d.%d",&a,&b,&c,&d) != 4) return false;
	if ((unsigned)a > 255 || (unsigned)b > 255 || (unsigned)c > 255 || (unsigned)d > 255) return false;
	out[0] = (uint8_t)a; out[1] = (uint8_t)b; out[2] = (uint8_t)c; out[3] = (uint8_t)d;
	return true;
}

void WebUI::handleClient(EthernetClient& c)
{
	char line[192];
	if (!readRequestLine(c,line,sizeof(line),1500)) { c.stop(); return; }

	drainHeaders(c,1500);

	bool isGet = startsWith(line,"GET ");
	bool isPost = startsWith(line,"POST ");
	if (!isGet && !isPost) { sendHeader(c,404,F("text/plain")); c.print(F("Bad method")); c.stop(); return; }

	const char* p = isGet ? (line + 4) :(line + 5);
	char path[128];
	size_t i = 0;
	while (*p && *p != ' ' && i + 1 < sizeof(path)) path[i++] = *p++;
	path[i] = 0;

	char* q = strchr(path,'?');
	char* query = nullptr;
	if (q) { *q = 0; query = q + 1; urlDecodeInPlace(query); }

	if (isGet && strcmp(path,"/") == 0) { pageHome(c); }
	else if (isGet && strcmp(path,"/internet1") == 0) { pageInternet1(c); }
	else if (isGet && strcmp(path,"/internet2") == 0) { pageInternet2(c); }
	else if (isGet && strcmp(path,"/gsm") == 0) { pageGsm(c); }
	else if (isPost && strcmp(path,"/internet1/save") == 0) { actionInternet1Save(c,query); }
	else if (isPost && strcmp(path,"/internet2/save") == 0) { actionInternet2Save(c,query); }
	else if (isPost && strcmp(path,"/gsm/save") == 0) { actionGsmSave(c,query); }
	else { pageNotFound(c); }

	c.stop();
}

{

		switch (*s) {
		case '&':c.print(F("&amp;")); break;

		case '>':c.print(F("&gt;")); break;

	}

bool WebUI::readRequestLine(EthernetClient& c,char* out,size_t outN,uint32_t timeoutMs)
{
	uint32_t t0 = millis();
	size_t n = 0;

	while ((millis() - t0) < timeoutMs) {
		while (c.available()) {
			char ch = (char)c.read();
			if (ch == '\r') continue;
			if (ch == '\n') {
				out[n] = 0;
				return (n > 0);
			}
			if (n + 1 < outN) out[n++] = ch;
		}
	}
	out[0] = 0;
	return false;
}

void WebUI::drainHeaders(EthernetClient& c,uint32_t timeoutMs)
{
	uint32_t t0 = millis();
	int crlfCount = 0;
	while ((millis() - t0) < timeoutMs) {
		while (c.available()) {
			char ch = (char)c.read();
			if (ch == '\n') {
				crlfCount++;
				if (crlfCount >= 2) return;
			}
			else if (ch != '\r') {
				crlfCount = 0;
			}
		}
	}
}

void WebUI::urlDecodeInPlace(char* s)
{
	auto hex = [](char c)->int {
		if (c >= '0' && c <= '9') return c - '0';
		if (c >= 'A' && c <= 'F') return 10 + (c - 'A');
		if (c >= 'a' && c <= 'f') return 10 + (c - 'a');
		return -1;
	};

	char* w = s;
	for (char* r = s; *r; r++) {
		if (*r == '+') { *w++ = ' '; continue; }
		if (*r == '%' && r[1] && r[2]) {
			int hi = hex(r[1]),lo = hex(r[2]);
			if (hi >= 0 && lo >= 0) {
				*w++ = (char)((hi << 4) | lo);
				r += 2;
				continue;
			}
		}
		*w++ = *r;
	}
	*w = 0;
}

const char* WebUI::findParam(char* query,const char* key)
{
	size_t klen = strlen(key);
	char* p = query;

	while (p && *p) {
		char* amp = strchr(p,'&');
		if (amp) *amp = 0;

		char* eq = strchr(p,'=');
		if (eq) {
			*eq = 0;
			if (strlen(p) == klen && strcmp(p,key) == 0) {
				char* val = eq + 1;
				if (amp) *amp = '&';
				*eq = '=';
				return val;
			}
			*eq = '=';
		}

		if (amp) {
			*amp = '&';
			p = amp + 1;
		}
		else break;
	}
	return nullptr;
}

bool WebUI::parseBool01(const char* s,bool& out)
{
	if (!s) return false;
	if (strcmp(s,"0") == 0) { out = false; return true; }
	if (strcmp(s,"1") == 0) { out = true; return true; }
	return false;
}

bool WebUI::parseU8(const char* s,uint8_t& out,uint8_t minV,uint8_t maxV)
{
	if (!s) return false;
	long v = strtol(s,nullptr,10);
	if (v < minV || v > maxV) return false;
	out = (uint8_t)v;
	return true;
}

bool WebUI::parseIp4(const char* s,uint8_t out[4])
{
	if (!s) return false;
	int a,b,c,d;
	if (sscanf(s,"%d.%d.%d.%d",&a,&b,&c,&d) != 4) return false;
	if ((unsigned)a > 255 || (unsigned)b > 255 || (unsigned)c > 255 || (unsigned)d > 255) return false;
	out[0] = (uint8_t)a; out[1] = (uint8_t)b; out[2] = (uint8_t)c; out[3] = (uint8_t)d;
	return true;
}

void WebUI::handleClient(EthernetClient& c)
{
	char line[192];
	if (!readRequestLine(c,line,sizeof(line),1500)) { c.stop(); return; }

	drainHeaders(c,1500);

	bool isGet = startsWith(line,"GET ");
	bool isPost = startsWith(line,"POST ");
	if (!isGet && !isPost) { sendHeader(c,404,F("text/plain")); c.print(F("Bad method")); c.stop(); return; }

	const char* p = isGet ? (line + 4) :(line + 5);
	char path[128];
	size_t i = 0;
	while (*p && *p != ' ' && i + 1 < sizeof(path)) path[i++] = *p++;
	path[i] = 0;

	char* q = strchr(path,'?');
	char* query = nullptr;
	if (q) { *q = 0; query = q + 1; urlDecodeInPlace(query); }

	if (isGet && strcmp(path,"/") == 0) { pageHome(c); }
	else if (isGet && strcmp(path,"/internet1") == 0) { pageInternet1(c); }
	else if (isGet && strcmp(path,"/internet2") == 0) { pageInternet2(c); }
	else if (isGet && strcmp(path,"/gsm") == 0) { pageGsm(c); }
	else if (isPost && strcmp(path,"/internet1/save") == 0) { actionInternet1Save(c,query); }
	else if (isPost && strcmp(path,"/internet2/save") == 0) { actionInternet2Save(c,query); }
	else if (isPost && strcmp(path,"/gsm/save") == 0) { actionGsmSave(c,query); }
	else { pageNotFound(c); }

	c.stop();
}

void WebUI::actionInternet1Save(EthernetClient& c,char* query)
{
	if (!query) { sendHeader(c,404,F("text/plain")); c.print(F("No query")); return; }

	LanIfStore::IfConfig cfg{};
	lanStore.load(LanIfStore::IF1,cfg,false);

	bool dhcp;
	uint8_t ntpIdx,perIdx;
	uint8_t ip[4],mask[4],gw[4],dns[4];

	if (parseBool01(findParam(query,"dhcp"),dhcp)) cfg.dhcp = dhcp ? 1 :0;
	if (parseU8(findParam(query,"ntpIdx"),ntpIdx,0,4)) cfg.ntpIdx = ntpIdx;
	if (parseU8(findParam(query,"perIdx"),perIdx,0,5)) cfg.periodIdx = perIdx;

	const char* sip = findParam(query,"ip");
	const char* smask = findParam(query,"mask");
	const char* sgw = findParam(query,"gw");
	const char* sdns = findParam(query,"dns");

	if (sip && parseIp4(sip,ip)) memcpy(cfg.ip,ip,4);
	if (smask && parseIp4(smask,mask)) memcpy(cfg.mask,mask,4);
	if (sgw && parseIp4(sgw,gw)) memcpy(cfg.gw,gw,4);
	if (sdns && parseIp4(sdns,dns)) memcpy(cfg.dns,dns,4);

	lanStore.save(LanIfStore::IF1,cfg,true);
	applyInternet1FromStore();

	sendRedirect(c,"/internet1");
}

ST‑LINK показывает,что микроконтроллер STM32F103VGT6 (XL‑density) с Flash = 1 MByte. Значит ошибка вида:

section .rodata will not fit in region FLASH
region FLASH overflowed ...
возникает не из‑за “реально не хватает памяти”,а потому что Arduino IDE/STM32 core собирает проект 
с линкер‑скриптом на 64/128/256К,т.е. выбран не тот размер Flash (или не тот “Board part number”).

Ниже что сделать.

1) Проверьте настройки платы в Arduino IDE (STM32duino)
Вы выбрали: Generic STM32F1 series (STMicroelectronics_GenF1) — это нормально,но внутри важно правильно выбрать вариант чипа/Flash.

Откройте Tools и выставьте:

Board part number:STM32F103VG / STM32F103VGT6 / F103VG (1MB) (название зависит от версии core)
Flash size:1MB (1024K) (если есть отдельный пункт)
Optimize:Smallest (-Os) (если есть)
LTO:Enabled (если есть)
Debug symbols:None (если есть)
Если пунктов Flash size / Board part number нет или там максимум 128/256К — значит у вас либо
 старая версия STM32 core,либо выбрана “не та” плата.

Что должно быть по сути
Линкер должен взять скрипт типа:

.../ldscript/stm32f103xg.ld (xG = 1MB) а не .../stm32f103xb.ld (128K) и не .../stm32f103xc.ld (256K).

#include "SyncSourcesStore.h"

SyncSourcesStore::SyncSourcesStore(TwoWire& w,uint8_t addr)
	:_w(&w),_addr(addr)
{}

bool SyncSourcesStore::begin()
{
	// Проверка наличия EEPROM на шине
	_w->beginTransmission(_addr);
	return (_w->endTransmission() == 0);
}

void SyncSourcesStore::defaults(Data& d)
{
	memset(&d,0,sizeof(d));

	d.gpsEnable = 0;
	d.gpsPeriodIdx = 3;

	d.netEnable = 0;
	d.netProviderIdx = 0;
	d.netPeriodIdx = 3;

	d.gsmEnable = 0;
	d.gsmProviderIdx = 0;
	d.gsmPeriodIdx = 3;

	d.net2Enable = 0;
	d.net2PeriodIdx = 3;
}

bool SyncSourcesStore::load(Data& d)
{
	// strict load from NEW BASE only
	return loadFromBase(BASE,d,true);
}

bool SyncSourcesStore::loadLoose(Data& d)
{
	// 1) Пытаемся читать из НОВОЙ области
	if (loadFromBase(BASE,d,true))
		return true;

	// 2) Если в новой области пусто/нет magic — пробуем СТАРУЮ (для миграции)
	Data tmp;
	if (loadFromBase(OLD_BASE,tmp,true))
	{
		// Миграция:сохраняем в новую область
		(void)saveToBase(BASE,tmp,true);
		d = tmp;
		return true;
	}

	// 3) Ничего валидного — defaults
	defaults(d);
	return false;
}

bool SyncSourcesStore::save(const Data& d)
{
	return saveToBase(BASE,d,true);
}

// ---------------------- low-level helpers ----------------------

bool SyncSourcesStore::readBytes(uint16_t memAddr,uint8_t* out,size_t len)
{
	// Типичный протокол AT24:записать адрес (2 байта),затем repeated-start и чтение
	_w->beginTransmission(_addr);
	_w->write((uint8_t)(memAddr >> 8));
	_w->write((uint8_t)(memAddr & 0xFF));
	if (_w->endTransmission(false) != 0) return false;

	size_t got = _w->requestFrom((int)_addr,(int)len);
	if (got != len) return false;

	for (size_t i = 0; i < len; i++)
	{
		int v = _w->read();
		if (v < 0) return false;
		out[i] = (uint8_t)v;
	}
	return true;
}

bool SyncSourcesStore::writeBytes(uint16_t memAddr,const uint8_t* data,size_t len)
{
	// Пишем постранично (PAGE_SIZE=64)
	size_t off = 0;
	while (off < len)
	{
		const uint16_t curAddr = (uint16_t)(memAddr + off);
		const uint16_t pageOff = (uint16_t)(curAddr % PAGE_SIZE);
		const size_t chunk = min((size_t)(PAGE_SIZE - pageOff),len - off);

		if (!writePage(curAddr,data + off,chunk)) return false;
		if (!waitReady(50)) return false;

		off += chunk;
	}
	return true;
}

bool SyncSourcesStore::writePage(uint16_t memAddr,const uint8_t* data,size_t len)
{
	_w->beginTransmission(_addr);
	_w->write((uint8_t)(memAddr >> 8));
	_w->write((uint8_t)(memAddr & 0xFF));
	for (size_t i = 0; i < len; i++) _w->write(data[i]);
	return (_w->endTransmission() == 0);
}

bool SyncSourcesStore::waitReady(uint32_t timeoutMs)
{
	const uint32_t t0 = millis();
	while ((millis() - t0) < timeoutMs)
	{
		_w->beginTransmission(_addr);
		if (_w->endTransmission() == 0) return true;
		delay(1);
	}
	return false;
}

bool SyncSourcesStore::readU8(uint16_t memAddr,uint8_t& v)
{
	return readBytes(memAddr,&v,1);
}

bool SyncSourcesStore::writeU8(uint16_t memAddr,uint8_t v)
{
	return writeBytes(memAddr,&v,1);
}

bool SyncSourcesStore::readU16(uint16_t memAddr,uint16_t& v)
{
	uint8_t b[2];
	if (!readBytes(memAddr,b,2)) return false;
	v = (uint16_t)((uint16_t)b[0] | ((uint16_t)b[1] << 8)); // LE
	return true;
}

bool SyncSourcesStore::writeU16(uint16_t memAddr,uint16_t v)
{
	uint8_t b[2];
	b[0] = (uint8_t)(v & 0xFF);
	b[1] = (uint8_t)(v >> 8);
	return writeBytes(memAddr,b,2);
}

// ---------------------- base-aware load/save (migration support) ----------------------

bool SyncSourcesStore::loadFromBase(uint16_t base,Data& d,bool strict)
{
	uint16_t mg = 0;
	uint8_t ver = 0;

	if (!readU16(base + 0,mg)) return false;
	if (!readU8(base + 2,ver)) return false;

	if (mg != MAGIC)
	{
		// если strict — сразу отказ
		// если loose — тоже отказ (т.к. мы не хотим "угадывать" данные)
		(void)strict;
		return false;
	}

	if (ver == 1)
	{
		// old V1:8 bytes (до gsmPeriodIdx включительно)
		uint8_t b[OLD_SIZE_V1];
		if (!readBytes(base + 4,b,OLD_SIZE_V1)) return false;

		memset(&d,0,sizeof(d));
		d.gpsEnable = b[0];
		d.gpsPeriodIdx = b[1];
		d.netEnable = b[2];
		d.netProviderIdx = b[3];
		d.netPeriodIdx = b[4];
		d.gsmEnable = b[5];
		d.gsmProviderIdx = b[6];
		d.gsmPeriodIdx = b[7];

		// новые поля V2
		d.net2Enable = 0;
		d.net2PeriodIdx = 3;
		return true;
	}
	else if (ver == 2)
	{
		// V2:полный размер структуры (10 байт)
		uint8_t b[sizeof(Data)];
		if (!readBytes(base + 4,b,sizeof(Data))) return false;
		memcpy(&d,b,sizeof(Data));
		return true;
	}

	// неизвестная версия
	return false;
}

bool SyncSourcesStore::saveToBase(uint16_t base,const Data& d,bool writeMagic)
{
	// MAGIC
	if (writeMagic)
	{
		if (!writeU16(base + 0,MAGIC)) return false;
	}

	// VERSION
	if (!writeU8(base + 2,VERSION)) return false;

	// DATA
	const uint8_t* p = reinterpret_cast<const uint8_t*>(&d);
	if (!writeBytes(base + 4,p,sizeof(Data))) return false;

	return true;
}

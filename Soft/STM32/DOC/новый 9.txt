	Ниже — вариант переделки Menu.h/Menu.cpp под вашу архитектуру:

Главная страница (HOME) постоянно обновляется (через Menu::update()),независимо от меню.
Каждая “страница” Str1..Str5 (то,что выбирается в MENU_PAGE_SELECT) является отдельной программой с методами:

void Menu::update()
{
 const uint32_t now = millis();

 // HOME постоянно обновляется,когда MENU_IDLE
 if (state == MENU_IDLE) {
 if (!_homeStaticDrawn) {
 drawHomeStatic();
 _homeStaticDrawn = true;
 _homeDirty = true;
 }

 // обновление динамики (время/сеть/сигнал) без полного fillScreen
 if (_homeDirty || (now - _lastHomeTickMs) > 200) {
 _lastHomeTickMs = now;
 drawHomeDynamic();
 _homeDirty = false;
 }
 return;
 }

 // активная страница обновляется сама
 if (state == MENU_ITEM_SELECT) {
 if (pages[page]) pages[page]->update();
 }
}

{

 tft->setTextColor(TFT_WHITE,TFT_BLACK);
 tft->drawString(_lastTime,80,80);

 tft->setTextColor(gNetRegistered ? TFT_GREEN :TFT_RED,TFT_BLACK);

 else snprintf(buf,sizeof(buf),"%ddBm [%u/5]",(int)gRssiDbm,(unsigned)gSignalBars);

 tft->drawString(buf,80,140);
 }

void Menu::onButtonPressed(int buttonID)
{
 // HOME:короткое SW1 — открыть меню
 if (state == MENU_IDLE) {
 if (buttonID == BTN_SW1) activate();
 return;
 }

 // PAGE SELECT:SW2/SW3 листают,SW1 фиксирует
 if (state == MENU_PAGE_SELECT) {
 if (buttonID == BTN_SW2) prevPage();
 else if (buttonID == BTN_SW3) nextPage();
 else if (buttonID == BTN_SW1) fixPage();
 return;
 }

 // ACTIVE PAGE:передаём странице + даём базовую навигацию
 if (state == MENU_ITEM_SELECT) {
 if (buttonID == BTN_SW2) prevItem();
 else if (buttonID == BTN_SW3) nextItem();
 else if (buttonID == BTN_SW1) backToStart();
 else if (buttonID == BTN_SW4) fixItem();

 if (pages[page]) pages[page]->onButtonPressed(buttonID);
 return;
 }
}

void Menu::onButtonReleased(int buttonID)
{
 if (state == MENU_ITEM_SELECT && pages[page]) pages[page]->onButtonReleased(buttonID);
}

Ниже — доработанный Button класс,который не ломает вашу текущую логику (BTN_SHORT/BTN_LONG через getEvent()),но добавляет:

события фронтов:нажал / отпустил (getEdge())

Это позволит вызывать в меню/страницах:

#ifndef BUTTONS_H
#define BUTTONS_H

#include <stdint.h>

enum ButtonEvent { BTN_NONE,BTN_SHORT,BTN_LONG };
enum ButtonEdge { EDGE_NONE,EDGE_PRESSED,EDGE_RELEASED };

class Button {
public:
 Button(uint8_t pin,unsigned long longPressMs = 700);

 // Вызывать в loop()
 void update();

 // Совместимо со старым кодом:события короткое/длинное нажатие (одноразовые)
 ButtonEvent getEvent();

 // Новое:фронты нажатия/отпускания (одноразовые)
 ButtonEdge getEdge();

 // Новое:текущее состояние
 bool isPressed() const;

private:
 uint8_t pin;

 bool lastLevel; // уровень пина на предыдущем update():true=HIGH (отпущено при INPUT_PULLUP)
 bool pressedFlag; // идет ли удержание
 bool longSent; // уже отправляли BTN_LONG в рамках текущего удержания

 unsigned long longPressDuration;

};

#endif
Buttons.cpp (обновлённый)
cpp Копировать
#include "Buttons.h"
#include <Arduino.h>

Button::Button(uint8_t apin,unsigned long longPressMs)
{
 pin = apin;
 pinMode(pin,INPUT_PULLUP);

 lastLevel = true; // HIGH = отпущено
 pressedFlag = false;
 longSent = false;

 pressedTime = 0;
 longPressDuration = longPressMs;

 event = BTN_NONE;
 edge = EDGE_NONE;
}

bool Button::isPressed() const
{
 // INPUT_PULLUP:LOW = нажато
 return digitalRead(pin) == LOW;
}

void Button::update()
{
 bool level = digitalRead(pin); // true=HIGH (отпущено),false=LOW (нажато)

 event = BTN_NONE;
 edge = EDGE_NONE;

 // фронт:отпущено->нажато
 if (!level && lastLevel)
 {
 edge = EDGE_PRESSED;
 pressedFlag = true;
 longSent = false;
 pressedTime = millis();
 }

 // фронт:нажато->отпущено
 if (level && !lastLevel)
 {
 edge = EDGE_RELEASED;

 if (pressedFlag)
 {
 unsigned long d = millis() - pressedTime;

 // короткое нажатие:только если long ещё не было
 if (!longSent && d > 15 && d < longPressDuration)
 {
 event = BTN_SHORT;
 }

 pressedFlag = false;
 longSent = false;
 }
 }

 // удержание:long press
 if (!level && pressedFlag && !longSent)
 {
 if ((millis() - pressedTime) >= longPressDuration)
 {
 event = BTN_LONG;
 longSent = true;
 // pressedFlag можно оставить true до отпускания
 }
 }

 lastLevel = level;
}

ButtonEvent Button::getEvent()
{
 return event;
}

ButtonEdge Button::getEdge()
{
 return edge;
}

cpp Копировать
#include "Buttons.h"
#include <Arduino.h>

Button::Button(uint8_t apin,unsigned long longPressMs,unsigned long debounceMs)
{
 pin = apin;
 pinMode(pin,INPUT_PULLUP);

 longPressDuration = longPressMs;
 debounceDuration = debounceMs;

 lastRawLevel = true;
 stableLevel = true;
 lastStableLevel = true;
 lastRawChangeMs = 0;

 pressedFlag = false;
 longSent = false;
 pressedTime = 0;
 lastReleaseDuration = 0;

 event = BTN_NONE;
 edge = EDGE_NONE;
}

bool Button::isPressed() const
{
 // stable (debounced) состояние
 return stableLevel == false; // INPUT_PULLUP:LOW=нажато
}

unsigned long Button::pressDurationMs() const
{
 if (isPressed()) {
 return millis() - pressedTime;
 }
 return lastReleaseDuration;
}

void Button::update()
{
 event = BTN_NONE;
 edge = EDGE_NONE;

 // 1) читаем raw уровень
 bool raw = digitalRead(pin); // true=HIGH (отпущено),false=LOW (нажато)
 unsigned long now = millis();

 // 2) отслеживаем изменение raw (для debounce)
 if (raw != lastRawLevel) {
 lastRawLevel = raw;
 lastRawChangeMs = now;
 }

 // 3) если raw стабилен debounceDuration — принимаем как stable
 if ((now - lastRawChangeMs) >= debounceDuration) {
 stableLevel = raw;
 }

 // 4) фронты по stable уровню
 if (stableLevel != lastStableLevel) {

 if (!stableLevel && lastStableLevel) {
 // отпущено -> нажато
 edge = EDGE_PRESSED;
 pressedFlag = true;
 longSent = false;
 pressedTime = now;
 }

 if (stableLevel && !lastStableLevel) {
 // нажато -> отпущено
 edge = EDGE_RELEASED;

 if (pressedFlag) {
 unsigned long d = now - pressedTime;
 lastReleaseDuration = d;

 // short:только если long ещё не отправляли
 if (!longSent && d < longPressDuration) {
 event = BTN_SHORT;
 }

 pressedFlag = false;
 longSent = false;
 } else {
 // если по какой-то причине отпустили без pressedFlag — всё равно фиксируем
 lastReleaseDuration = 0;
 }
 }

 lastStableLevel = stableLevel;
 }

 // 5) long press (по stable удержанию)
 if (!stableLevel && pressedFlag && !longSent) {
 if ((now - pressedTime) >= longPressDuration) {
 event = BTN_LONG;
 longSent = true;
 }
 }
}

ButtonEvent Button::getEvent() { return event; }
ButtonEdge Button::getEdge() { return edge; }

void Menu::update()
{
 const uint32_t now = millis();

 // HOME постоянно обновляется
 if (state == MENU_IDLE) {
 if (!_homeStaticDrawn) {
 drawHomeStatic();
 _homeStaticDrawn = true;
 _homeDirty = true;
 }

 // обновлять динамику не чаще 200 мс
 if (_homeDirty || (now - _lastHomeTickMs) > 200) {

 return;

 if (state == MENU_ITEM_SELECT) {
 if (pages[page]) pages[page]->update();
 }
}

Нужно разделить Page2 на:

static (рисуется один раз при onActivate()):заголовки/подписи/рамки
dynamic (обновляется без fillScreen):только значения TZ и индикатор “SAVED”,и только если они изменились.
Ниже — готовая замена Page2,которая перерисовывает только изменившиеся области через fillRect(...) + drawString(...)
 и использует кэш предыдущих значений.

Код предполагает,что Page2 остаётся внутри Menu.cpp как класс.

Page2 (обновлённая,частичная перерисовка)
cpp Копировать
class Page2 :public PageBase {
public:
 using PageBase::PageBase;

 void setup() override {}

 void onActivate() override {
 _staticDrawn = false;
 _lastTzTarget = 127;
 _lastTzNtp = 127;
 _lastSavedVisible = false;
 _savedMsgMs = 0;

 draw(); // нарисует static+dynamic
 }

 void update() override {
 // скрыть "SAVED" через 1 сек
 if (_savedMsgMs && (millis() - _savedMsgMs) > 1000) {
 _savedMsgMs = 0;
 drawDynamic(); // только динамику
 }
 }

 void draw() override {
 if (!_staticDrawn) {
 drawStatic();
 _staticDrawn = true;
 }
 drawDynamic();
 }

 void onButtonPressed(int buttonID) override {
 if (buttonID == BTN_SW2) {
 cfg.tzTargetHours--;
 clampTz(cfg.tzTargetHours);
 drawDynamic();
 }
 else if (buttonID == BTN_SW3) {
 cfg.tzTargetHours++;
 clampTz(cfg.tzTargetHours);
 drawDynamic();
 }
 else if (buttonID == BTN_SW4) {
 ee.writeTzTargetHours(cfg.tzTargetHours);
 _savedMsgMs = millis();
 drawDynamic();
 }
 }

 void onButtonReleased(int) override {}

private:
 bool _staticDrawn = false;

 int8_t _lastTzTarget = 127;
 int8_t _lastTzNtp = 127;

 bool _lastSavedVisible = false;
 uint32_t _savedMsgMs = 0;

 // координаты областей (подберите под вашу верстку)
 static const int X_LABEL = 10;
 static const int X_VAL = 170;
 static const int Y_TZT = 60;
 static const int Y_TZN = 90;
 static const int W_VAL = 140;
 static const int H_VAL = 22;

 static const int X_SAVED = 240;
 static const int Y_SAVED = 190;
 static const int W_SAVED = 80;
 static const int H_SAVED = 22;

 void drawStatic() {
 TFT_eSPI* d = tft();
 if (!d) return;

 d->fillScreen(TFT_BLACK);

 d->setTextSize(2);
 d->setTextColor(TFT_YELLOW,TFT_BLACK);
 d->drawString("PAGE 2:TIME ZONE",10,10);

 d->setTextColor(TFT_WHITE,TFT_BLACK);
 d->drawString("TZ target:",X_LABEL,Y_TZT);
 d->drawString("TZ NTP :",X_LABEL,Y_TZN);

 d->drawString("SW2 - TZ-",10,140);
 d->drawString("SW3 - TZ+",10,165);
 d->drawString("SW4 - SAVE",10,190);
 d->drawString("SW1 - HOME",10,215);

 // очистим области динамики
 d->fillRect(X_VAL,Y_TZT,W_VAL,H_VAL,TFT_BLACK);
 d->fillRect(X_VAL,Y_TZN,W_VAL,H_VAL,TFT_BLACK);
 d->fillRect(X_SAVED,Y_SAVED,W_SAVED,H_SAVED,TFT_BLACK);
 }

 void drawDynamic() {
 TFT_eSPI* d = tft();
 if (!d) return;

 // TZ target (перерисовать только если изменился)
 if (_lastTzTarget != cfg.tzTargetHours) {
 _lastTzTarget = cfg.tzTargetHours;
 d->fillRect(X_VAL,Y_TZT,W_VAL,H_VAL,TFT_BLACK);

 char buf[16];
 snprintf(buf,sizeof(buf),"%d",(int)_lastTzTarget);
 d->setTextColor(TFT_WHITE,TFT_BLACK);
 d->drawString(buf,X_VAL,Y_TZT);
 }

 // TZ NTP (если вы позже будете его менять — готово)
 if (_lastTzNtp != cfg.tzNtpHours) {
 _lastTzNtp = cfg.tzNtpHours;
 d->fillRect(X_VAL,Y_TZN,W_VAL,H_VAL,TFT_BLACK);

 char buf[16];
 snprintf(buf,sizeof(buf),"%d",(int)_lastTzNtp);
 d->setTextColor(TFT_WHITE,TFT_BLACK);
 d->drawString(buf,X_VAL,Y_TZN);
 }

 // SAVED indicator
 bool savedVisible = (_savedMsgMs != 0);
 if (_lastSavedVisible != savedVisible) {
 _lastSavedVisible = savedVisible;
 d->fillRect(X_SAVED,Y_SAVED,W_SAVED,H_SAVED,TFT_BLACK);
 if (savedVisible) {
 d->setTextColor(TFT_GREEN,TFT_BLACK);
 d->drawString("SAVED",X_SAVED,Y_SAVED);
 }
 }
 }

 static void clampTz(int8_t& tz) {
 if (tz < -12) tz = -12;
 if (tz > 14) tz = 14;
 }
};
Что это даёт
При входе на страницу рисуется экран один раз (fillScreen только в drawStatic()).
При изменении TZ перерисовывается только прямоугольник со значением.
“SAVED” рисуется/стирается только при изменении видимости.
Рекомендация (чтобы не было мерцания)

Если вы используете loadFont() — делайте это в drawStatic(),а в drawDynamic() не меняйте шрифт.

Если покажете вашу фактическую разметку страницы 2 (координаты/шрифты),я подставлю размеры 
прямоугольников под ваш шрифт так,чтобы точно затиралось значение без “хвостов”.

Причина “время на главной обновляется редко” почти всегда такая:на HOME вы рисуете gTimeStr,а 
gTimeStr обновляется только когда пришло время от SIM800 (sim.hasNewTime()),то есть раз в минуту/10 минут — поэтому на экране “редко”.

Чтобы HOME обновлялся каждые 500 мс,нужно периодически обновлять gTimeStr/gDateStr из RTC,а не ждать SIM800.

1) Обновлять gTimeStr/gDateStr из RTC каждые 500 мс (в .ino)
Добавьте в loop() (или отдельной функцией),например сразу после RTCTime now = rtc.getTime();:

cpp Копировать
// Обновлять строки времени для дисплея каждые 500 мс
static uint32_t lastUiTimeMs = 0;
if (millis() - lastUiTimeMs >= 500) {
 lastUiTimeMs += 500;

 RTCTime now = rtc.getTime();

 snprintf(gDateStr,sizeof(gDateStr),"%02u.%02u.%04u",
 (unsigned)now.dayOfMonth,(unsigned)now.month,(unsigned)now.year);
 snprintf(gTimeStr,sizeof(gTimeStr),"%02u:%02u:%02u",
 (unsigned)now.hour,(unsigned)now.minute,(unsigned)now.second);

 gTimeUpdated = true; // если используете этот флаг
 menu_start.invalidateHome(); // чтобы HOME перерисовал динамику сразу
}
Теперь даже если SIM800 обновляет RTC редко,HOME будет показывать “живые” секунды.

cpp Копировать
class Page1 :public PageBase {
public:
 using PageBase::PageBase;

 void setup() override {}

 void onActivate() override {
 _mode = MODE_LIST;
 _sel = 0;
 _staticDrawn = false;
 draw();
 }

 void update() override {
 // тут можно обновлять нижнюю строку по спутникам,если MODE_GPS
 if (_mode == MODE_GPS) drawGpsSatInfo(false);
 }

 void draw() override {
 if (!_staticDrawn) {
 drawStatic();
 _staticDrawn = true;
 }
 drawDynamic(true);
 }

 void onButtonPressed(int buttonID) override {
 if (_mode == MODE_LIST) {
 if (buttonID == BTN_SW2) { _sel = (_sel + 1) % 5; drawDynamic(false); }
 else if (buttonID == BTN_SW3) { _sel = (_sel + 4) % 5; drawDynamic(false); }
 else if (buttonID == BTN_SW4) {
 if (_sel == 0) { _mode = MODE_GPS; _subSel = 0; _staticDrawn=false; draw(); }
 else if (_sel == 1) { _mode = MODE_NET; _subSel = 0; _staticDrawn=false; draw(); }
 else if (_sel == 2) { _mode = MODE_GSM; _subSel = 0; _staticDrawn=false; draw(); }
 }
 else if (buttonID == BTN_SW1) {
 _m->backToStart(); // HOME
 }
 return;
 }

 // Подменю GPS/NET/GSM
 if (buttonID == BTN_SW1) { // назад в список
 _mode = MODE_LIST;
 _staticDrawn = false;
 draw();
 return;
 }

 // навигация по пунктам подменю
 if (buttonID == BTN_SW2) { _subSel = (_subSel + 1) % subCount(); drawDynamic(false); return; }
 if (buttonID == BTN_SW3) { _subSel = (_subSel + subCount() - 1) % subCount(); drawDynamic(false); return; }

 // SW4:изменить значение выбранного подпункта
 if (buttonID == BTN_SW4) {
 applyActionOnSubItem();
 drawDynamic(false);
 return;
 }
 }

 void onButtonReleased(int) override {}

private:
 enum Mode { MODE_LIST,MODE_GPS,MODE_NET,MODE_GSM } _mode = MODE_LIST;
 bool _staticDrawn = false;

 uint8_t _sel = 0; // выбор источника в списке
 uint8_t _subSel = 0; // выбор подпункта в подменю

 // кэши,чтобы не перерисовывать лишнее (упрощённо)
 uint8_t _lastSel = 255;
 uint8_t _lastSubSel = 255;

 void drawStatic() {
 auto d = tft(); if(!d) return;
 d->fillScreen(TFT_BLACK);
 d->setTextSize(2);

 if (_mode == MODE_LIST) {
 d->setTextColor(TFT_YELLOW,TFT_BLACK);
 d->drawString("ИСТОЧНИКИ СИНХРОНИЗАЦИИ",10,10);
 } else if (_mode == MODE_GPS) {
 d->setTextColor(TFT_YELLOW,TFT_BLACK);
 d->drawString("GPS",10,10);
 } else if (_mode == MODE_NET) {
 d->setTextColor(TFT_YELLOW,TFT_BLACK);
 d->drawString("ИНТЕРНЕТ",10,10);
 } else if (_mode == MODE_GSM) {
 d->setTextColor(TFT_YELLOW,TFT_BLACK);
 d->drawString("GSM",10,10);
 }

 // очистка областей строк
 d->setTextColor(TFT_WHITE,TFT_BLACK);
 if (_mode == MODE_LIST) {
 for (int i=0;i<5;i++) d->fillRect(10,50+i*28,300,24,TFT_BLACK);
 } else {
 // подменю (до 3 пунктов),плюс нижняя строка у GPS
 for (int i=0;i<4;i++) d->fillRect(10,50+i*28,300,24,TFT_BLACK);
 }
 _lastSel = 255;
 _lastSubSel = 255;
 }

 void drawDynamic(bool force) {
 auto d = tft(); if(!d) return;

 if (_mode == MODE_LIST) {
 if (!force && _lastSel == _sel) return;
 _lastSel = _sel;

 for (int i=0;i<5;i++) {
 d->fillRect(10,50+i*28,300,24,TFT_BLACK);
 d->setTextColor(i==_sel ? TFT_GREEN :TFT_WHITE,TFT_BLACK);
 d->drawString(_m->itemLabel(0,i),10,50+i*28);
 }
 return;
 }

 // подменю
 if (!force && _lastSubSel == _subSel) {
 // всё равно у GPS обновим инфо по спутникам
 if (_mode == MODE_GPS) drawGpsSatInfo(false);
 return;
 }
 _lastSubSel = _subSel;

 drawSubMenuLines();
 if (_mode == MODE_GPS) drawGpsSatInfo(true);
 }

 uint8_t subCount() const {
 if (_mode == MODE_GPS) return 2;
 if (_mode == MODE_NET) return 3;
 if (_mode == MODE_GSM) return 3;
 return 0;
 }

 void drawSubMenuLines() {
 auto d = tft(); if(!d) return;

 // 0:Enable,1:Period,2:Provider (если есть)
 for (int i=0;i<subCount();i++) {
 d->fillRect(10,50+i*28,300,24,TFT_BLACK);
 d->setTextColor(i==_subSel ? TFT_GREEN :TFT_WHITE,TFT_BLACK);

 char line[64];
 if (_mode == MODE_GPS) {
 if (i==0) snprintf(line,sizeof(line),"1) ВКЛ:%s",gSync.gpsEnable ? "ДА":"НЕТ");
 if (i==1) snprintf(line,sizeof(line),"2) ПЕРИОД:%s",kPeriodsName[gSync.gpsPeriodIdx]);
 }
 else if (_mode == MODE_NET) {
 if (i==0) snprintf(line,sizeof(line),"1) ВКЛ:%s",gSync.netEnable ? "ДА":"НЕТ");
 if (i==1) snprintf(line,sizeof(line),"2) ПРОВ:%s",kNetProviders[gSync.netProviderIdx]);
 if (i==2) snprintf(line,sizeof(line),"3) ПЕРИОД:%s",kPeriodsName[gSync.netPeriodIdx]);
 }
 else if (_mode == MODE_GSM) {
 if (i==0) snprintf(line,sizeof(line),"1) ВКЛ:%s",gSync.gsmEnable ? "ДА":"НЕТ");
 if (i==1) snprintf(line,sizeof(line),"2) ПРОВ:%s",kGsmProviders[gSync.gsmProviderIdx]);
 if (i==2) snprintf(line,sizeof(line),"3) ПЕРИОД:%s",kPeriodsName[gSync.gsmPeriodIdx]);
 }

 d->drawString(line,10,50+i*28);
 }
 }

 void drawGpsSatInfo(bool force) {
 auto d = tft(); if(!d) return;
 const int y = 50 + 3*28; // нижняя строка
 if (force) d->fillRect(10,y,300,24,TFT_BLACK);
 d->setTextColor(TFT_CYAN,TFT_BLACK);

 char buf[64];
 snprintf(buf,sizeof(buf),"SAT:%u/%u FIX:%s",
 (unsigned)gGpsSatsUsed,(unsigned)gGpsSatsView,gGpsFix ? "YES":"NO");
 d->drawString(buf,10,y);
 }

 void applyActionOnSubItem() {
 if (_mode == MODE_GPS) {
 if (_subSel == 0) gSync.gpsEnable = !gSync.gpsEnable;
 if (_subSel == 1) gSync.gpsPeriodIdx = (gSync.gpsPeriodIdx + 1) % 6;
 }
 else if (_mode == MODE_NET) {
 if (_subSel == 0) gSync.netEnable = !gSync.netEnable;
 if (_subSel == 1) gSync.netProviderIdx = (gSync.netProviderIdx + 1) % 5;
 if (_subSel == 2) gSync.netPeriodIdx = (gSync.netPeriodIdx + 1) % 6;
 }
 else if (_mode == MODE_GSM) {
 if (_subSel == 0) gSync.gsmEnable = !gSync.gsmEnable;
 if (_subSel == 1) gSync.gsmProviderIdx = (gSync.gsmProviderIdx + 1) % 5;
 if (_subSel == 2) gSync.gsmPeriodIdx = (gSync.gsmPeriodIdx + 1) % 6;
 }
 // Здесь же можно сразу сохранять в EEPROM по SW4 (или сделать сохранение отдельным пунктом)
 }
};
6) Сохранение в AT24C128 (фиксированные адреса)
Вы просили “записывать в память”. Для этих настроек предлагаю зарезервировать область,например с 0x0200.

Минимальная карта:

0x0200 gpsEnable (u8)
0x0201 gpsPeriodIdx (u8)
0x0202 netEnable (u8)
0x0203 netProviderIdx (u8)
0x0204 netPeriodIdx (u8)
0x0205 gsmEnable (u8)
0x0206 gsmProviderIdx (u8)
0x0207 gsmPeriodIdx (u8)
Чтобы это сделать аккуратно,лучше добавить в AT24C128Settings методы readU8/writeU8 публично или 
сделать отдельный небольшой класс SyncSettingsStore 
(как вы делали для AT24C128Settings). Я могу дать готовые файлы .h/.cpp под это.

#pragma once
#include <Arduino.h>
#include <Wire.h>

class SyncSourcesStore {
public:
 static constexpr uint8_t I2C_ADDR = 0x50;

 struct Data {
 uint8_t gpsEnable; // 0/1
 uint8_t gpsPeriodIdx; // 0..5

 uint8_t netEnable; // 0/1
 uint8_t netProviderIdx;// 0..4 (NTP server index)
 uint8_t netPeriodIdx; // 0..5

 uint8_t gsmEnable; // 0/1
 uint8_t gsmProviderIdx;// 0..4 (оператор)
 uint8_t gsmPeriodIdx; // 0..5
 };

 explicit SyncSourcesStore(TwoWire& w = Wire,uint8_t addr = I2C_ADDR);

 bool begin();
 void defaults(Data& d);
 bool load(Data& d); // strict (magic/ver)
 bool loadLoose(Data& d);// best effort (если magic нет — defaults)
 bool save(const Data& d);

 // фикс. адреса (для внешних программ)
 static constexpr uint16_t BASE = 0x0200;
 static constexpr uint16_t ADDR_MAGIC = BASE + 0; // u16
 static constexpr uint16_t ADDR_VER = BASE + 2; // u8
 static constexpr uint16_t ADDR_DATA = BASE + 4; // Data bytes

private:
 TwoWire* _w;
 uint8_t _addr;

 static constexpr uint16_t MAGIC = 0x5359; // 'SY'
 static constexpr uint8_t VERSION = 1;
 static constexpr uint8_t PAGE_SIZE = 64;

 bool readBytes(uint16_t memAddr,uint8_t* out,size_t len);
 bool writeBytes(uint16_t memAddr,const uint8_t* data,size_t len);
 bool writePage(uint16_t memAddr,const uint8_t* data,size_t len);
 bool waitReady(uint32_t timeoutMs = 50);

 bool readU8(uint16_t memAddr,uint8_t& v);
 bool writeU8(uint16_t memAddr,uint8_t v);
 bool readU16(uint16_t memAddr,uint16_t& v);
 bool writeU16(uint16_t memAddr,uint16_t v);
};
SyncSourcesStore.cpp (Arduino_STM32/libmaple)
cpp Копировать
#include "SyncSourcesStore.h"

SyncSourcesStore::SyncSourcesStore(TwoWire& w,uint8_t addr) :_w(&w),_addr(addr) {}

bool SyncSourcesStore::begin() {
 _w->beginTransmission(_addr);
 return (_w->endTransmission() == 0);
}

void SyncSourcesStore::defaults(Data& d) {
 d.gpsEnable = 0; d.gpsPeriodIdx = 0;
 d.netEnable = 0; d.netProviderIdx = 0; d.netPeriodIdx = 0;
 d.gsmEnable = 1; d.gsmProviderIdx = 0; d.gsmPeriodIdx = 0;
}

bool SyncSourcesStore::load(Data& d) {
 defaults(d);
 uint16_t mg = 0; uint8_t ver = 0;
 if (!readU16(ADDR_MAGIC,mg) || mg != MAGIC) return false;
 if (!readU8(ADDR_VER,ver) || ver != VERSION) return false;
 return readBytes(ADDR_DATA,(uint8_t*)&d,sizeof(d));
}

bool SyncSourcesStore::loadLoose(Data& d) {
 if (!load(d)) { defaults(d); return false; }
 return true;
}

bool SyncSourcesStore::save(const Data& d) {
 if (!writeU16(ADDR_MAGIC,MAGIC)) return false;
 if (!writeU8(ADDR_VER,VERSION)) return false;
 return writeBytes(ADDR_DATA,(const uint8_t*)&d,sizeof(d));
}

// ---- low level (libmaple) ----
bool SyncSourcesStore::readBytes(uint16_t memAddr,uint8_t* out,size_t len) {
 _w->beginTransmission(_addr);
 _w->write((uint8_t)(memAddr >> 8));
 _w->write((uint8_t)(memAddr & 0xFF));
 if (_w->endTransmission() != 0) return false;

 uint8_t got = _w->requestFrom((uint8_t)_addr,(int)len);
 if (got != (uint8_t)len) return false;

 for (size_t i=0;i<len;i++) out[i] = (uint8_t)_w->read();
 return true;
}

bool SyncSourcesStore::writeBytes(uint16_t memAddr,const uint8_t* data,size_t len) {
 size_t off = 0;
 while (off < len) {
 uint16_t a = memAddr + (uint16_t)off;
 uint8_t pageOff = (uint8_t)(a % PAGE_SIZE);
 size_t chunk = min((size_t)(PAGE_SIZE - pageOff),len - off);
 if (!writePage(a,data + off,chunk)) return false;
 if (!waitReady(50)) return false;
 off += chunk;
 }
 return true;
}

bool SyncSourcesStore::writePage(uint16_t memAddr,const uint8_t* data,size_t len) {
 _w->beginTransmission(_addr);
 _w->write((uint8_t)(memAddr >> 8));
 _w->write((uint8_t)(memAddr & 0xFF));
 _w->write((uint8_t*)data,(int)len); // libmaple требует non-const
 return (_w->endTransmission() == 0);
}

bool SyncSourcesStore::waitReady(uint32_t timeoutMs) {
 uint32_t t0 = millis();
 while ((millis() - t0) < timeoutMs) {
 _w->beginTransmission(_addr);
 if (_w->endTransmission() == 0) return true;
 delay(1);
 }
 return false;
}

bool SyncSourcesStore::readU8(uint16_t memAddr,uint8_t& v) { return readBytes(memAddr,&v,1); }
bool SyncSourcesStore::writeU8(uint16_t memAddr,uint8_t v) { return writeBytes(memAddr,&v,1); }

bool SyncSourcesStore::readU16(uint16_t memAddr,uint16_t& v) {
 uint8_t b[2]; if(!readBytes(memAddr,b,2)) return false;
 v = (uint16_t)b[0] | ((uint16_t)b[1] << 8);
 return true;
}
bool SyncSourcesStore::writeU16(uint16_t memAddr,uint16_t v) {
 uint8_t b[2] = {(uint8_t)(v & 0xFF),(uint8_t)((v >> 8) & 0xFF)};
 return writeBytes(memAddr,b,2);
}

Planner в .ino будет запускать Internet/GSM синхронизацию по расписанию (как мы сделали в TimeSyncPlanner).
Минимальный чек‑лист (чтобы не “undefined reference”)

SyncSourcesStore.h/.cpp
TimeSyncPlanner.h/.cpp
В .ino объекты глобальные:
cpp Копировать
SyncSourcesStore syncStore(Wire,0x50);
SyncSourcesStore::Data syncData;
TimeSyncPlanner planner(sim,ee,cfg,syncStore,syncData);
В setup():
cpp Копировать
syncStore.begin();
syncStore.loadLoose(syncData);
planner.begin();
В loop():
cpp Копировать
sim.tick();
planner.tick();
if (sim.hasNewTime()) planner.onTimeUpdated();

Ниже — полностью готовые файлы TimeSyncPlanner.h / TimeSyncPlanner.cpp под ваш текущий проект/ядро 
(Arduino_STM32,STM32F103,Arduino IDE). Планировщик использует ваши глобальные объекты:

SIM800TimeAsync sim;
AT24C128Settings ee; AT24C128Settings::Config cfg;
SyncSourcesStore syncStore; SyncSourcesStore::Data syncData;
и обеспечивает приоритет GPS > Internet(NTP) > GSM(CCLK),с режимом «догонять пропущенное» 
(не запускает много раз подряд,а переносит “следующий запуск” в будущее).

Важно:для Internet(NTP) планировщик вызывает sim.requestNtpNow(). Убедитесь,что в вашем 
SIM800TimeAsync добавлен метод requestNtpNow() (я ранее описывал патч).

TimeSyncPlanner.h
cpp Копировать
#pragma once
#include <Arduino.h>

#include "SIM800TimeAsync.h"
#include "AT24C128Settings.h"
#include "SyncSourcesStore.h"

// Планировщик синхронизации времени по источникам (GPS > Internet > GSM).
// - GPS пока не реализован:просто "пропускается",но расписание ведется.
// - Internet:принудительный NTP (AT+CNTP),затем CCLK и запись в RTC (у вас в .ino).
// - GSM:только CCLK (NITZ/RTC модема).
//
// "Догонять пропущенное":если время запуска уже прошло,следующий запуск переносится вперед,
// и выполняется не серия запусков,а один.
class TimeSyncPlanner {
public:
 TimeSyncPlanner(SIM800TimeAsync& sim,
 AT24C128Settings& eeCfg,
 AT24C128Settings::Config& cfg,
 SyncSourcesStore& srcStore,
 SyncSourcesStore::Data& src);

 void begin();

 // Вызывать в loop() часто
 void tick();

 // Вызывать после изменения настроек из меню (после SAVE)
 void onSettingsChanged();

 // Вызывать при получении нового времени от SIM800 (после sim.hasNewTime())
 void onTimeUpdated();

 const char* lastAction() const { return _lastAction; }
 uint32_t lastActionMs() const { return _lastActionMs; }

private:
 SIM800TimeAsync& _sim;
 AT24C128Settings& _eeCfg;
 AT24C128Settings::Config& _cfg;
 SyncSourcesStore& _srcStore;
 SyncSourcesStore::Data& _src;

 uint32_t _nextGpsMs = 0;
 uint32_t _nextNetMs = 0;
 uint32_t _nextGsmMs = 0;

 bool _netInFlight = false;
 bool _gsmInFlight = false;

 // GPS пока не реализован
 bool _gpsImplemented = false;

 const char* _lastAction = "none";
 uint32_t _lastActionMs = 0;

private:
 static uint32_t periodMsFromIdx(uint8_t idx);

 // переносит next в будущее (>now) с учетом period
 static uint32_t catchUpNext(uint32_t now,uint32_t next,uint32_t period);

 void applyNtpConfig(bool enableFallback);

 bool due(uint32_t now,uint32_t next) const { return (int32_t)(now - next) >= 0; }

 bool tryStartGps(uint32_t now);
 bool tryStartNet(uint32_t now);
 bool tryStartGsm(uint32_t now);
};
TimeSyncPlanner.cpp
cpp Копировать
#include "TimeSyncPlanner.h"

TimeSyncPlanner::TimeSyncPlanner(SIM800TimeAsync& sim,
 AT24C128Settings& eeCfg,
 AT24C128Settings::Config& cfg,
 SyncSourcesStore& srcStore,
 SyncSourcesStore::Data& src)
:_sim(sim),_eeCfg(eeCfg),_cfg(cfg),_srcStore(srcStore),_src(src) {}

uint32_t TimeSyncPlanner::periodMsFromIdx(uint8_t idx)
{
 // индексы должны совпадать с меню:1м,10м,30м,1ч,6ч,12ч
 static const uint32_t kPeriodsMs[6] = {
 60000UL,
 600000UL,
 1800000UL,
 3600000UL,
 21600000UL,
 43200000UL
 };
 return kPeriodsMs[idx % 6];
}

uint32_t TimeSyncPlanner::catchUpNext(uint32_t now,uint32_t next,uint32_t period)
{
 if (period == 0) return now + 60000UL;
 if ((int32_t)(now - next) < 0) return next;

 // next уже в прошлом:пересчитать так,чтобы next стало > now
 uint32_t diff = now - next;
 uint32_t steps = diff / period + 1;
 return next + steps * period;
}

void TimeSyncPlanner::applyNtpConfig(bool enableFallback)
{
 // NTP параметры берём из cfg (server/apn/user/pass/tzNtpHours)
 SIM800TimeAsync::NtpConfig ntp;
 ntp.apn = _cfg.apn;
 ntp.user = _cfg.user;
 ntp.pass = _cfg.pass;
 ntp.server = _cfg.server;
 ntp.tzHours = _cfg.tzNtpHours; // обычно 0 (UTC)
 ntp.enableFallback = enableFallback;
 _sim.setNtpConfig(ntp);
}

void TimeSyncPlanner::begin()
{
 const uint32_t now = millis();

 _nextGpsMs = now + periodMsFromIdx(_src.gpsPeriodIdx);
 _nextNetMs = now + periodMsFromIdx(_src.netPeriodIdx);
 _nextGsmMs = now + periodMsFromIdx(_src.gsmPeriodIdx);

 // Рекомендация:чтобы SIM800TimeAsync сам не запускал периодический опрос времени,
 // можно "задвинуть" его период далеко:
 // _sim.setPeriodMs(86400000UL); // 1 сутки
}

void TimeSyncPlanner::onSettingsChanged()
{
 // когда пользователь меняет периоды,важно не получить "лавину" запусков
 const uint32_t now = millis();
 _nextGpsMs = catchUpNext(now,_nextGpsMs,periodMsFromIdx(_src.gpsPeriodIdx));
 _nextNetMs = catchUpNext(now,_nextNetMs,periodMsFromIdx(_src.netPeriodIdx));
 _nextGsmMs = catchUpNext(now,_nextGsmMs,periodMsFromIdx(_src.gsmPeriodIdx));
}

void TimeSyncPlanner::onTimeUpdated()
{
 _netInFlight = false;
 _gsmInFlight = false;
}

bool TimeSyncPlanner::tryStartGps(uint32_t now)
{
 if (!_src.gpsEnable) return false;
 if (!due(now,_nextGpsMs)) return false;

 const uint32_t p = periodMsFromIdx(_src.gpsPeriodIdx);
 _nextGpsMs = catchUpNext(now,_nextGpsMs,p);

 if (!_gpsImplemented) {
 _lastAction = "gps(skip)";
 _lastActionMs = now;
 return false; // не блокируем,даём Internet/GSM продолжить
 }

 // Когда появится GPS:запуск тут.
 return false;
}

bool TimeSyncPlanner::tryStartNet(uint32_t now)
{
 if (!_src.netEnable) return false;
 if (!due(now,_nextNetMs)) return false;
 if (_sim.isBusy()) return false;

 const uint32_t p = periodMsFromIdx(_src.netPeriodIdx);
 _nextNetMs = catchUpNext(now,_nextNetMs,p);

 applyNtpConfig(true);

 // Internet источник = принудительный NTP
 _sim.requestNtpNow();
 _netInFlight = true;

 _lastAction = "internet(ntp)";
 _lastActionMs = now;
 return true;
}

bool TimeSyncPlanner::tryStartGsm(uint32_t now)
{
 if (!_src.gsmEnable) return false;
 if (!due(now,_nextGsmMs)) return false;
 if (_sim.isBusy()) return false;

 const uint32_t p = periodMsFromIdx(_src.gsmPeriodIdx);
 _nextGsmMs = catchUpNext(now,_nextGsmMs,p);

 // GSM источник = CCLK (без NTP)
 applyNtpConfig(false);

 _sim.requestTimeNow();
 _gsmInFlight = true;

 _lastAction = "gsm(cclk)";
 _lastActionMs = now;
 return true;
}

void TimeSyncPlanner::tick()
{
 const uint32_t now = millis();

 // Приоритет:GPS > Internet > GSM
 (void)tryStartGps(now);

 if (tryStartNet(now)) return;
 (void)tryStartGsm(now);
}

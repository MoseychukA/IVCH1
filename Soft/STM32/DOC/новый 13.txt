// Планировщик синхронизации времени (GPS > Internet(NTP) > GSM(CCLK)).
// "Догонять пропущенное":если next в прошлом,переносим next вперед,выполняем 1 запуск.
// Немедленный запуск:triggerImmediate() -> старт по приоритету в tick(),когда SIM свободен.
class TimeSyncPlanner {

 void tick();

 // После sim.hasNewTime() и записи в DS3231
 void onTimeUpdated();

 // Немедленно запустить синхронизацию (по приоритету). Не блокирует.
 void triggerImmediate();

 const char* lastAction() const { return _lastAction; }
 uint32_t lastActionMs() const { return _lastActionMs; }

private:
 SIM800TimeAsync& _sim;
 AT24C128Settings& _eeCfg;
 AT24C128Settings::Config& _cfg;
 SyncSourcesStore& _srcStore;

 uint32_t _nextGpsMs = 0;
 uint32_t _nextNetMs = 0;
 uint32_t _nextGsmMs = 0;

 bool _gpsImplemented = false;

 bool _immediatePending = false;

 const char* _lastAction = "none";

private:

uint32_t TimeSyncPlanner::periodMsFromIdx(uint8_t idx)
{
 static const uint32_t kPeriodsMs[6] = {
 60000UL,// 1 мин
 600000UL,// 10 мин
 1800000UL,// 30 мин
 3600000UL,// 1 час
 21600000UL,// 6 часов
 43200000UL // 12 часов
 };
 return kPeriodsMs[idx % 6];
}

uint32_t TimeSyncPlanner::catchUpNext(uint32_t now,uint32_t next,uint32_t period)
{
 if (period == 0) return now + 60000UL;
 if ((int32_t)(now - next) < 0) return next;

 uint32_t diff = now - next;
 uint32_t steps = diff / period + 1;
 return next + steps * period;
}

void TimeSyncPlanner::applyNtpConfig(bool enableFallback)
{
 SIM800TimeAsync::NtpConfig ntp;
 ntp.apn = _cfg.apn;
 ntp.user = _cfg.user;
 ntp.pass = _cfg.pass;
 ntp.server = _cfg.server; // выбирается в меню INTERNET (через cfg.server)
 ntp.tzHours = _cfg.tzNtpHours; // обычно 0 (UTC)
 ntp.enableFallback = enableFallback;

 _sim.setNtpConfig(ntp);
}

void TimeSyncPlanner::begin()
{
 uint32_t now = millis();
 _nextGpsMs = now + periodMsFromIdx(_src.gpsPeriodIdx);
 _nextNetMs = now + periodMsFromIdx(_src.netPeriodIdx);
 _nextGsmMs = now + periodMsFromIdx(_src.gsmPeriodIdx);
}

void TimeSyncPlanner::onSettingsChanged()
{
 uint32_t now = millis();
 _nextGpsMs = catchUpNext(now,_nextGpsMs,periodMsFromIdx(_src.gpsPeriodIdx));
 _nextNetMs = catchUpNext(now,_nextNetMs,periodMsFromIdx(_src.netPeriodIdx));
 _nextGsmMs = catchUpNext(now,_nextGsmMs,periodMsFromIdx(_src.gsmPeriodIdx));
}

bool TimeSyncPlanner::tryStartGps(uint32_t now)
{
 if (!_src.gpsEnable) return false;

 // немедленно:попытка без проверки расписания
 // по расписанию:проверяем due()
 // Мы различаем это по _immediatePending (в tick),поэтому здесь оставим обычную логику due.
 if (!due(now,_nextGpsMs)) return false;

 uint32_t p = periodMsFromIdx(_src.gpsPeriodIdx);
 _nextGpsMs = catchUpNext(now,_nextGpsMs,p);

 if (!_gpsImplemented) {
 _lastAction = "gps(skip)";
 _lastActionMs = now;
 return false;
 }

 // TODO:GPS запуск
 return false;
}

bool TimeSyncPlanner::tryStartNet(uint32_t now)
{
 if (!_src.netEnable) return false;
 if (_sim.isBusy()) return false;

 // по расписанию
 if (!due(now,_nextNetMs)) return false;

 uint32_t p = periodMsFromIdx(_src.netPeriodIdx);
 _nextNetMs = catchUpNext(now,_nextNetMs,p);

 applyNtpConfig(true);

 // Internet = принудительный NTP
 _sim.requestNtpNow();

 _lastAction = "internet(ntp)";
 _lastActionMs = now;
 return true;
}

bool TimeSyncPlanner::tryStartGsm(uint32_t now)
{
 if (!_src.gsmEnable) return false;
 if (_sim.isBusy()) return false;

 if (!due(now,_nextGsmMs)) return false;

 uint32_t p = periodMsFromIdx(_src.gsmPeriodIdx);
 _nextGsmMs = catchUpNext(now,_nextGsmMs,p);

 applyNtpConfig(false);

 _sim.requestTimeNow();

 _lastAction = "gsm(cclk)";
 _lastActionMs = now;
 return true;
}

void TimeSyncPlanner::tick()
{
 uint32_t now = millis();

 // Немедленный запуск:выполнить как только возможно (SIM свободен и источник включен)
 if (_immediatePending) {

 // Для немедленного запуска мы НЕ хотим ждать _nextX — запускаем приоритетом по enabled.
 // Поэтому делаем отдельные проверки без due() и сразу переносим next вперед.

 // 1) GPS (если реализован)
 if (_src.gpsEnable) {
 // если GPS не реализован — считаем "пропуск" и идем дальше
 uint32_t p = periodMsFromIdx(_src.gpsPeriodIdx);
 _nextGpsMs = now + p;

 if (_gpsImplemented) {
 // TODO:GPS start
 }
 _lastAction = "gps(immediate-skip)";
 _lastActionMs = now;
 // не return — GPS пока нет,дадим шанс NET/GSM
 }

 // 2) INTERNET (NTP)
 if (_src.netEnable && !_sim.isBusy()) {
 uint32_t p = periodMsFromIdx(_src.netPeriodIdx);
 _nextNetMs = now + p;

 applyNtpConfig(true);
 _sim.requestNtpNow();

 _lastAction = "internet(immediate-ntp)";
 _lastActionMs = now;
 _immediatePending = false;
 return;
 }

 // 3) GSM (CCLK)
 if (_src.gsmEnable && !_sim.isBusy()) {
 uint32_t p = periodMsFromIdx(_src.gsmPeriodIdx);
 _nextGsmMs = now + p;

 applyNtpConfig(false);
 _sim.requestTimeNow();

 _lastAction = "gsm(immediate-cclk)";
 _lastActionMs = now;
 _immediatePending = false;
 return;
 }

 // Если не смогли стартовать (SIM busy или источники выключены) — попробуем позже
 return;
 }

 // Обычное расписание:GPS > Internet > GSM
 (void)tryStartGps(now);
 if (tryStartNet(now)) return;
 (void)tryStartGsm(now);
}

int16_t SIM800TimeAsync::rssiDbm() const {
 if (_csqRssi == 99 || _csqRssi < 0 || _csqRssi > 31) return 0;
 return (int16_t)(-113 + 2 * _csqRssi);
}

uint8_t SIM800TimeAsync::signalBars() const {
 if (_csqRssi == 99) return 0;
 if (_csqRssi <= 2) return 0;
 if (_csqRssi <= 9) return 1;
 if (_csqRssi <= 14) return 2;
 if (_csqRssi <= 19) return 3;
 if (_csqRssi <= 25) return 4;
 return 5;
}

// ---- FSM ----
void SIM800TimeAsync::_enter(State s) {
 // reset AT transaction
 _atPending = false;
 _atExpect = nullptr;
 _atResp = "";
 _atDeadlineMs = 0;
 _atOk = false;
 _atErr = false;

 _st = s;
 _stEnterMs = millis();

 if (_dbg) {
 _dbg->print('['); _dbg->print(_stEnterMs); _dbg->print("] ");
 _dbg->print(F("STATE -> "));
 _dbg->println(stateName());
 }
}

void SIM800TimeAsync::_serviceState() {
 const uint32_t now = millis();

 // IDLE scheduler
 if (_st == State::IDLE) {

 // 1) health ping
 if (_healthPingMs > 0 && (int32_t)(now - _nextPingMs) >= 0) {
 _nextPingMs = now + _healthPingMs;
 _enter(State::AT_PING);
 return;
 }

 // 2) time poll
 if (_forcePoll || (int32_t)(now - _nextPollMs) >= 0) {
 _forcePoll = false;
 _nextPollMs = now + _periodMs;
 _enter(State::WAIT_NETWORK);
 return;
 }

 // 3) status poll
 if (_statusPollMs > 0 && (int32_t)(now - _nextStatusMs) >= 0) {
 _nextStatusMs = now + _statusPollMs;
 _enter(State::STAT_CREG);
 return;
 }

 return;
 }

 switch (_st) {

 case State::STOPPED:
 // auto retry every 60s
 if ((now - _stEnterMs) > 60000) {
 _recoveryUsed = 0;
 _enter(State::PROBE_AT);
 }
 return;

 // ---- health ----
 case State::AT_PING:
 if (!_atPending) {
 _startAT("AT","OK",1500);
 } else {
 if (_atOk) {
 _recoveryUsed = 0;
 _enter(State::IDLE);
 } else if (_atDoneTimeout()) {
 if (_recoveryUsed < _maxRecoveryAttempts) {
 _recoveryUsed++;
 _enter(State::PWRKEY_PULSE);
 } else {
 _enter(State::STOPPED);
 }
 }
 }
 return;

 // ---- status ----
 case State::STAT_CREG:
 if (!_atPending) {
 _startAT("AT+CREG?","+CREG:",1500);
 } else {
 if (_atMatchedExpect()) {
 int p = _atResp.indexOf("+CREG:");
 if (p >= 0) {
 int comma = _atResp.indexOf(',',p);
 if (comma >= 0) {
 int stat = _atResp.substring(comma + 1).toInt();
 if (stat >= 0 && stat <= 5) _cregStat = (uint8_t)stat;
 }
 }
 _enter(State::STAT_CSQ);
 } else if (_atDoneTimeout()) {
 _enter(State::STAT_CSQ);
 }
 }
 return;

 case State::STAT_CSQ:
 if (!_atPending) {
 _startAT("AT+CSQ","+CSQ:",1500);
 } else {
 if (_atMatchedExpect()) {
 int p = _atResp.indexOf("+CSQ:");
 if (p >= 0) {
 int colon = _atResp.indexOf(':',p);
 int comma = _atResp.indexOf(',',p);
 if (colon >= 0 && comma >= 0 && comma > colon) {
 int rssi = _atResp.substring(colon + 1,comma).toInt();
 int ber = _atResp.substring(comma + 1).toInt();
 _csqRssi = (int8_t)rssi;
 _csqBer = (uint8_t)ber;
 }
 }
 _enter(State::IDLE);
 } else if (_atDoneTimeout()) {
 _enter(State::IDLE);
 }
 }
 return;

 // ---- bring-up / init ----
 case State::PROBE_AT:
 if (!_atPending) {
 _startAT("AT","OK",1500);
 } else {
 if (_atOk) {
 _recoveryUsed = 0;
 _enter(State::INIT_ATE0);
 } else if (_atDoneTimeout()) {
 if (_recoveryUsed < _maxRecoveryAttempts) {
 _recoveryUsed++;
 _enter(State::PWRKEY_PULSE);
 } else {
 _enter(State::STOPPED);
 }
 }
 }
 return;

 case State::PWRKEY_PULSE:
 if (now - _stEnterMs < 50) _pwrkeyIdle();
 else if (now - _stEnterMs < 1300) _pwrkeyPress();
 else {
 _pwrkeyRelease();
 _seenRDY = _seenCallReady = _seenSMSReady = false;
 _enter(State::WAIT_URC);
 }
 return;

 case State::WAIT_URC:
 if (_seenRDY || _seenCallReady || _seenSMSReady || (now - _stEnterMs) > 8000) {
 _enter(State::WAIT_AT_READY);
 }
 return;

 case State::WAIT_AT_READY:
 if (!_atPending) {
 _startAT("AT","OK",12000);
 } else {
 if (_atOk) {
 _recoveryUsed = 0;
 _enter(State::INIT_ATE0);
 } else if (_atDoneTimeout()) {
 if (_recoveryUsed < _maxRecoveryAttempts) {
 _recoveryUsed++;
 _enter(State::PWRKEY_PULSE);
 } else {
 _enter(State::STOPPED);
 }
 }
 }
 return;

 case State::INIT_ATE0:
 if (!_atPending) _startAT("ATE0","OK",1500);
 else if (_atOk || _atDoneTimeout()) _enter(State::INIT_CMEE);
 return;

 case State::INIT_CMEE:
 if (!_atPending) _startAT("AT+CMEE=2","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::INIT_CLTS);
 return;

 case State::INIT_CLTS:
 if (!_atPending) _startAT("AT+CLTS=1","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::INIT_SAVE);
 return;

 case State::INIT_SAVE:
 if (!_atPending) _startAT("AT&W","OK",3000);
 else if (_atOk || _atDoneTimeout()) {
 _recoveryUsed = 0;
 _enter(State::IDLE);
 }
 return;

 // ---- time flow ----
 case State::WAIT_NETWORK:
 if (!_atPending) {
 _startAT("AT+CREG?","+CREG:",1500);
 } else {
 if (_atDoneTimeout()) {
 if (now - _stEnterMs > 60000) _enter(State::IDLE);
 else {
 _atPending = false;
 _atResp = "";
 _atOk = _atErr = false;
 }
 } else if (_atMatchedExpect()) {

 // cache CREG from this answer too
 int p = _atResp.indexOf("+CREG:");
 if (p >= 0) {
 int comma = _atResp.indexOf(',',p);
 if (comma >= 0) {
 int stat = _atResp.substring(comma + 1).toInt();
 if (stat >= 0 && stat <= 5) _cregStat = (uint8_t)stat;
 }
 }

 if (networkRegistered()) {
 // forced NTP?
 if (_forceNtp && _ntp.enableFallback && _ntp.apn && _ntp.apn[0]) {
 _forceNtp = false; // одноразово
 _enter(State::NTP_CGATT);
 } else {
 // если принудительный NTP был запрошен,но NTP не настроен - сбросим
 _forceNtp = false;
 _enter(State::REQ_CCLK);
 }
 } else {
 _atPending = false;
 _atResp = "";
 _atOk = _atErr = false;
 }
 }
 }
 return;

 case State::REQ_CCLK:
 if (!_atPending) {
 _lastCCLK = "";
 _startAT("AT+CCLK?","+CCLK:",2000);
 } else {
 if (_atMatchedExpect()) {
 if (_lastCCLK.length() == 0) _lastCCLK = _atResp; // fallback
 if (_lastCCLK.indexOf("+CCLK:") >= 0) _newTime = true;

 if (!_isCCLKValid(_lastCCLK,2020) && _ntp.enableFallback && _ntp.apn && _ntp.apn[0]) {
 _enter(State::NTP_CGATT);
 } else {
 _enter(State::IDLE);
 }
 } else if (_atDoneTimeout()) {
 _enter(State::IDLE);
 }
 }
 return;

 // ---- NTP branch ----
 case State::NTP_CGATT:
 if (!_atPending) _startAT("AT+CGATT?","+CGATT:",2000);
 else {
 if (_atDoneTimeout()) { _enter(State::NTP_SAPBR_CLOSE); return; }
 if (_atMatchedExpect()) {
 if (_atResp.indexOf("+CGATT:1") >= 0) _enter(State::NTP_SAPBR_CFG1);
 else { _atPending = false; _startAT("AT+CGATT=1","OK",5000); }
 }
 }
 return;

 case State::NTP_SAPBR_CFG1:
 if (!_atPending) _startAT("AT+SAPBR=3,1,\"Contype\",\"GPRS\"","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::NTP_SAPBR_CFG_APN);
 return;

 case State::NTP_SAPBR_CFG_APN:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"APN\",\"") + _ntp.apn + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter((_ntp.user && _ntp.user[0]) ? State::NTP_SAPBR_CFG_USER
 :((_ntp.pass && _ntp.pass[0]) ? State::NTP_SAPBR_CFG_PWD :State::NTP_SAPBR_OPEN));
 }
 return;
 }

 case State::NTP_SAPBR_CFG_USER:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"USER\",\"") + _ntp.user + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter((_ntp.pass && _ntp.pass[0]) ? State::NTP_SAPBR_CFG_PWD :State::NTP_SAPBR_OPEN);
 }
 return;
 }

 case State::NTP_SAPBR_CFG_PWD:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"PWD\",\"") + _ntp.pass + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter(State::NTP_SAPBR_OPEN);
 }
 return;
 }

 case State::NTP_SAPBR_OPEN:
 if (!_atPending) _startAT("AT+SAPBR=1,1","OK",45000);
 else if (_atOk || _atDoneTimeout()) _enter(State::NTP_CNTPCID);
 return;

 case State::NTP_CNTPCID:
 if (!_atPending) _startAT("AT+CNTPCID=1","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::NTP_CNTP_SET);
 return;

 case State::NTP_CNTP_SET:{
 if (!_atPending) {
 String cmd = String("AT+CNTP=\"") + (_ntp.server ? _ntp.server :"pool.ntp.org") + "\"," + String(_ntp.tzHours);
 _startAT(cmd.c_str(),"OK",3000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter(State::NTP_CNTP_START);
 }
 return;
 }

 case State::NTP_CNTP_START:
 if (!_atPending) _startAT("AT+CNTP","+CNTP:",20000);
 else {
 if (_atMatchedExpect() || _atDoneTimeout()) _enter(State::NTP_SAPBR_CLOSE);
 }
 return;

 case State::NTP_SAPBR_CLOSE:
 if (!_atPending) _startAT("AT+SAPBR=0,1","OK",8000);
 else if (_atOk || _atDoneTimeout()) _enter(State::REQ_CCLK_AFTER_NTP);
 return;

 case State::REQ_CCLK_AFTER_NTP:
 if (!_atPending) {
 _lastCCLK = "";
 _startAT("AT+CCLK?","+CCLK:",2000);
 } else {
 if (_atMatchedExpect()) {
 if (_lastCCLK.indexOf("+CCLK:") >= 0) _newTime = true;
 }
 _forceNtp = false; // на всякий случай
 _enter(State::IDLE);
 }
 return;
 }
}

// ---- UART ----
void SIM800TimeAsync::_readUartNonBlocking() {
 while (_sim.available()) {
 char c = (char)_sim.read();
 if (c == '\r') continue;

 if (c == '\n') {
 if (_lineBuf.length()) {
 _handleLine(_lineBuf);
 _lineBuf = "";
 }
 } else {
 if (_lineBuf.length() < 160) _lineBuf += c;
 }
 }
}

void SIM800TimeAsync::_handleLine(const String& line) {
 // URC hints
 if (line.indexOf("RDY") >= 0) _seenRDY = true;
 if (line.indexOf("Call Ready") >= 0) _seenCallReady = true;
 if (line.indexOf("SMS Ready") >= 0) _seenSMSReady = true;

 if (_dbg && _logAT) {
 _dbg->print('['); _dbg->print(millis()); _dbg->print("] ");
 _dbg->print(F("<< "));
 _dbg->println(line);
 }

 if (_atPending) {
 if (line == "OK") { _atOk = true; _lastAtOkMs = millis(); }
 else if (line == "ERROR") _atErr = true;

 if (line.startsWith("+CCLK:")) _lastCCLK = line;

 _atResp += line;
 _atResp += "\n";
 }
}

// ---- AT helpers ----
void SIM800TimeAsync::_startAT(const char* cmd,const char* expect,uint32_t timeoutMs) {
 _atResp = "";
 _atExpect = expect;
 _atPending = true;
 _atDeadlineMs = millis() + timeoutMs;
 _atOk = false;
 _atErr = false;

 if (_dbg && _logAT) {
 _dbg->print('['); _dbg->print(millis()); _dbg->print("] ");
 _dbg->print(F(">> "));
 _dbg->println(cmd);
 }

 _sim.print(cmd);
 _sim.print("\r");
}

bool SIM800TimeAsync::_atDoneTimeout() const {
 if (!_atPending) return false;
 return (int32_t)(millis() - _atDeadlineMs) >= 0;
}

bool SIM800TimeAsync::_atMatchedExpect() const {
 if (!_atPending || !_atExpect) return false;
 return _atResp.indexOf(_atExpect) >= 0;
}

// ---- time parse/valid ----
bool SIM800TimeAsync::parseCCLK(const String& resp,
 int& year,int& month,int& day,
 int& hour,int& minute,int& second,
 int& tzQuarterHours) {

 int q1 = resp.indexOf('\"');
 int q2 = resp.indexOf('\"',q1 + 1);
 if (q1 < 0 || q2 < 0) return false;

 String v = resp.substring(q1 + 1,q2); // yy/MM/dd,hh:mm:ss±zz
 if (v.length() < 17) return false;

 int yy = v.substring(0,2).toInt();
 month = v.substring(3,5).toInt();
 day = v.substring(6,8).toInt();
 hour = v.substring(9,11).toInt();
 minute = v.substring(12,14).toInt();
 second = v.substring(15,17).toInt();
 year = 2000 + yy;

 tzQuarterHours = 0;
 if (v.length() >= 20) {
 char sign = v.charAt(17);
 int zz = v.substring(18).toInt();
 tzQuarterHours = (sign == '-') ? -zz :zz;
 }
 return true;
}

bool SIM800TimeAsync::_isCCLKValid(const String& resp,int minYear) {
 int y,mo,d,h,mi,s,tzq;
 if (!parseCCLK(resp,y,mo,d,h,mi,s,tzq)) return false;
 if (y < minYear) return false;
 if (mo < 1 || mo > 12) return false;
 if (d < 1 || d > 31) return false;
 if (h < 0 || h > 23) return false;
 if (mi < 0 || mi > 59) return false;
 if (s < 0 || s > 59) return false;
 return true;
}

#include "SIM800Time.h"

SIM800TimeClient::SIM800TimeClient(HardwareSerial& serial,uint8_t pwrkeyPin,bool pwrkeyActiveHigh)
:_sim(serial),_pwrkeyPin(pwrkeyPin),_pwrkeyActiveHigh(pwrkeyActiveHigh) {}

void SIM800TimeClient::begin(uint32_t baud) {
 pinMode(_pwrkeyPin,OUTPUT);
 _setPWRKEYIdle(); // для NPN:idle обычно LOW (транзистор закрыт)
 _sim.begin(baud);
}

void SIM800TimeClient::_setPWRKEYIdle() {
 digitalWrite(_pwrkeyPin,_pwrkeyActiveHigh ? LOW :HIGH);
}

void SIM800TimeClient::_setPWRKEYPressed() {
 digitalWrite(_pwrkeyPin,_pwrkeyActiveHigh ? HIGH :LOW);
}

void SIM800TimeClient::_pulsePWRKEY(uint16_t pressMs,uint16_t preDelayMs) {
 _setPWRKEYIdle();
 delay(preDelayMs);
 _setPWRKEYPressed();
 delay(pressMs);
 _setPWRKEYIdle();
}

String SIM800TimeClient::_readAll(uint32_t timeoutMs) {
 String s;
 uint32_t t0 = millis();
 while (millis() - t0 < timeoutMs) {
 while (_sim.available()) s += (char)_sim.read();
 delay(2);
 }
 return s;
}

bool SIM800TimeClient::_sendAT(const char* cmd,const char* expect,uint32_t timeoutMs,String* out) {
 while (_sim.available()) _sim.read();

 _sim.print(cmd);
 _sim.print("\r");

 String resp = _readAll(timeoutMs);
 if (out) *out = resp;

 if (!expect) return true;
 return resp.indexOf(expect) >= 0;
}

bool SIM800TimeClient::_waitATReady(uint32_t timeoutMs) {
 uint32_t t0 = millis();
 while (millis() - t0 < timeoutMs) {
 if (_sendAT("AT","OK",700)) return true;
 delay(300);
 }
 return false;
}

// ---------- URC handling ----------
void SIM800TimeClient::_resetURCFlags() {
 _seenRDY = _seenCallReady = _seenSMSReady = false;
}

void SIM800TimeClient::_drainInput(uint32_t ms) {
 uint32_t t0 = millis();
 while (millis() - t0 < ms) {
 while (_sim.available()) (void)_sim.read();
 delay(2);
 }
}

// Простейший line-scanner:ловим RDY/Call Ready/SMS Ready в потоке
void SIM800TimeClient::_pollURC(uint32_t ms) {
 String line;
 uint32_t t0 = millis();
 while (millis() - t0 < ms) {
 while (_sim.available()) {
 char c = (char)_sim.read();
 if (c == '\r') continue;
 if (c == '\n') {
 if (line.length()) {
 if (line.indexOf("RDY") >= 0) _seenRDY = true;
 if (line.indexOf("Call Ready") >= 0) _seenCallReady = true;
 if (line.indexOf("SMS Ready") >= 0) _seenSMSReady = true;
 line = "";
 }
 } else {
 // ограничим рост строки
 if (line.length() < 96) line += c;
 }
 }
 delay(2);
 }
}

bool SIM800TimeClient::ensureOnSmart(uint32_t atProbeTimeoutMs,
 uint16_t pwrkeyPressMs,
 uint32_t urcWaitMs,
 uint32_t atReadyTimeoutMs) {
 // Если уже отвечает на AT — не трогаем PWRKEY (чтобы случайно не выключить)
 if (_sendAT("AT","OK",atProbeTimeoutMs)) return true;

 _resetURCFlags();
 _drainInput(200);

 // Пытаемся включить
 _pulsePWRKEY(pwrkeyPressMs,200);

 // Подождать URC загрузки
 _pollURC(urcWaitMs);

 // Независимо от URC — финальная проверка AT
 return _waitATReady(atReadyTimeoutMs);
}

bool SIM800TimeClient::init(uint32_t atReadyTimeoutMs) {
 if (!_waitATReady(atReadyTimeoutMs)) return false;

 _sendAT("ATE0","OK",800);
 _sendAT("AT+CMEE=2","OK",800);

 // NITZ
 _sendAT("AT+CLTS=1","OK",1200);
 _sendAT("AT&W","OK",1200);

 return true;
}

bool SIM800TimeClient::waitForNetwork(uint32_t timeoutMs) {
 uint32_t t0 = millis();
 while (millis() - t0 < timeoutMs) {
 String r;
 if (_sendAT("AT+CREG?","+CREG:",1200,&r)) {
 int idx = r.indexOf("+CREG:");
 if (idx >= 0) {
 if (r.indexOf(",1",idx) >= 0 || r.indexOf(",5",idx) >= 0) return true;
 }
 }
 delay(1000);
 }
 return false;
}

bool SIM800TimeClient::getCCLK(String& rawResp,uint32_t timeoutMs) {
 return _sendAT("AT+CCLK?","+CCLK:",timeoutMs,&rawResp);
}

bool SIM800TimeClient::parseCCLK(const String& resp,
 int& year,int& month,int& day,
 int& hour,int& minute,int& second,
 int& tzQuarterHours) {
 int q1 = resp.indexOf('\"');
 int q2 = resp.indexOf('\"',q1 + 1);
 if (q1 < 0 || q2 < 0) return false;

 String v = resp.substring(q1 + 1,q2); // yy/MM/dd,hh:mm:ss±zz
 if (v.length() < 17) return false;

 int yy = v.substring(0,2).toInt();
 month = v.substring(3,5).toInt();
 day = v.substring(6,8).toInt();
 hour = v.substring(9,11).toInt();
 minute = v.substring(12,14).toInt();
 second = v.substring(15,17).toInt();
 year = 2000 + yy;

 tzQuarterHours = 0;
 if (v.length() >= 20) {
 char sign = v.charAt(17);
 int zz = v.substring(18).toInt(); // четверти часа
 tzQuarterHours = (sign == '-') ? -zz :zz;
 }
 return true;
}

bool SIM800TimeClient::isCCLKValid(const String& resp,int minYear) {
 int y,mo,d,h,mi,s,tzq;
 if (!parseCCLK(resp,y,mo,d,h,mi,s,tzq)) return false;
 if (y < minYear) return false;
 if (mo < 1 || mo > 12) return false;
 if (d < 1 || d > 31) return false;
 if (h < 0 || h > 23) return false;
 if (mi < 0 || mi > 59) return false;
 if (s < 0 || s > 59) return false;
 return true;
}

// ---------- GPRS / NTP ----------
bool SIM800TimeClient::_waitCGATT(uint32_t timeoutMs) {
 uint32_t t0 = millis();
 while (millis() - t0 < timeoutMs) {
 String r;
 if (_sendAT("AT+CGATT?","+CGATT:",1200,&r)) {
 // +CGATT:1
 if (r.indexOf("+CGATT:1") >= 0) return true;
 }
 // пробуем приаттачиться
 _sendAT("AT+CGATT=1","OK",3000);
 delay(1000);
 }
 return false;
}

bool SIM800TimeClient::_bearerClose() {
 // Закрыть,если был открыт (OK/ERROR не критично)
 _sendAT("AT+SAPBR=0,1","OK",5000);
 return true;
}

bool SIM800TimeClient::_bearerOpen(const char* apn,const char* user,const char* pass,uint32_t timeoutMs) {
 _bearerClose();

 if (!_sendAT("AT+SAPBR=3,1,\"Contype\",\"GPRS\"","OK",2000)) return false;

 // APN обязателен
 {
 String cmd = String("AT+SAPBR=3,1,\"APN\",\"") + apn + "\"";
 if (!_sendAT(cmd.c_str(),"OK",2000)) return false;
 }

 // user/pass опциональны
 if (user && user[0]) {
 String cmd = String("AT+SAPBR=3,1,\"USER\",\"") + user + "\"";
 _sendAT(cmd.c_str(),"OK",2000);
 }
 if (pass && pass[0]) {
 String cmd = String("AT+SAPBR=3,1,\"PWD\",\"") + pass + "\"";
 _sendAT(cmd.c_str(),"OK",2000);
 }

 // Открыть bearer
 if (!_sendAT("AT+SAPBR=1,1","OK",timeoutMs)) return false;

 // Проверить IP
 String r;
 if (!_sendAT("AT+SAPBR=2,1","+SAPBR:",3000,&r)) return false;
 // Обычно:+SAPBR:1,1,"10.160.57.123"
 if (r.indexOf(",1,") < 0 && r.indexOf(",1,\"") < 0) {
 // не всегда строго,но попытка проверки
 }
 return true;
}

bool SIM800TimeClient::_ntpRequest(const char* server,int tzHours,uint32_t timeoutMs) {
 // Привязать NTP к bearer profile 1
 if (!_sendAT("AT+CNTPCID=1","OK",2000)) return false;

 // AT+CNTP="pool.ntp.org",<tz>
 {
 String cmd = String("AT+CNTP=\"") + server + "\"," + String(tzHours);
 if (!_sendAT(cmd.c_str(),"OK",3000)) return false;
 }

 // Запуск NTP:ответ приходит как OK и потом +CNTP:<result>
 // result:1=успех (часто),другие значения — ошибка (DNS/timeout/etc)
 while (_sim.available()) _sim.read();
 _sim.print("AT+CNTP\r");

 String resp = _readAll(timeoutMs);
 // Ищем +CNTP:1
 int p = resp.indexOf("+CNTP:");
 if (p < 0) return false;

 // вытащим число после +CNTP:
 int comma = resp.indexOf(':',p);
 if (comma < 0) return false;
 int end = resp.indexOf('\n',comma);
 if (end < 0) end = resp.length();
 String tail = resp.substring(comma + 1,end);
 tail.trim();
 int code = tail.toInt();

 return (code == 1);
}

bool SIM800TimeClient::syncTimeWithNTP(const char* apn,
 const char* user,
 const char* pass,
 const char* ntpServer,
 int tzHours,
 uint32_t bearerOpenTimeoutMs,
 uint32_t ntpTimeoutMs) {
 // Должны быть в сети,и желательно attached
 if (!_waitCGATT(30000)) return false;

 if (!_bearerOpen(apn,user,pass,bearerOpenTimeoutMs)) {
 _bearerClose();
 return false;
 }

 bool ok = _ntpRequest(ntpServer,tzHours,ntpTimeoutMs);

 // Можно закрыть bearer,чтобы не держать сессию
 _bearerClose();

 return ok;
}

bool SIM800TimeClient::getTimeSmart(String& rawResp,
 const char* apn,
 const char* user,
 const char* pass,
 const char* ntpServer,
 int tzHours,
 bool forceNtpIfInvalid) {
 if (!waitForNetwork(60000)) {
 rawResp = "No network registration (+CREG)";
 return false;
 }

 // Попробовать NITZ/RTC
 delay(1500);
 if (!getCCLK(rawResp,1500)) return false;

 if (isCCLKValid(rawResp)) return true;
 if (!forceNtpIfInvalid) return true; // вернули как есть (невалидное),если так нужно

 // Fallback NTP
 if (!syncTimeWithNTP(apn,user,pass,ntpServer,tzHours)) {
 // не смогли синхронизировать — вернем исходный rawResp
 return true;
 }

 // После NTP модем обновляет RTC,читаем снова
 delay(500);
 return getCCLK(rawResp,1500);
}

#include <Arduino.h>
#include "SIM800Time.h"

// UART4 на STM32F103VGT6:PC10(TX),PC11(RX)
static const PinName SIM_TX = PC10;
static const PinName SIM_RX = PC11;

static const uint32_t SIM_BAUD = 9600;

// PWRKEY через NPN на землю:"нажать" = PE0 HIGH
static const uint8_t SIM_PWRKEY_PIN = PE0;
static const bool PWRKEY_ACTIVE_HIGH = true;

HardwareSerial simSerial(SIM_RX,SIM_TX); // RX,TX
SIM800TimeClient sim(simSerial,SIM_PWRKEY_PIN,PWRKEY_ACTIVE_HIGH);

// Настройки оператора (заполните!)
static const char* APN = "internet"; // пример
static const char* USER = ""; // часто пусто
static const char* PASS = ""; // часто пусто

// NTP сервер
static const char* NTP_SERVER = "pool.ntp.org";

// Таймзона для AT+CNTP в ЧАСАХ (не четверти часа)
static const int TZ_HOURS = 3; // например MSK = +3

static uint32_t tLast = 0;
static const uint32_t PERIOD_MS = 60000;

static void printTime(const String& resp) {
 int y,mo,d,h,mi,s,tzq;
 if (SIM800TimeClient::parseCCLK(resp,y,mo,d,h,mi,s,tzq)) {
 Serial.print("CCLK:");
 Serial.print(y); Serial.print("-");
 if (mo < 10) Serial.print("0"); Serial.print(mo); Serial.print("-");
 if (d < 10) Serial.print("0"); Serial.print(d); Serial.print(" ");
 if (h < 10) Serial.print("0"); Serial.print(h); Serial.print(":");
 if (mi < 10) Serial.print("0"); Serial.print(mi); Serial.print(":");
 if (s < 10) Serial.print("0"); Serial.print(s);
 Serial.print(" TZ(qhr)=");
 Serial.print(tzq);

 Serial.print(" valid=");
 Serial.println(SIM800TimeClient::isCCLKValid(resp) ? "yes" :"no");
 } else {
 Serial.print("Raw:");
 Serial.println(resp);
 }
}

void setup() {
 Serial.begin(115200);
 delay(200);

 Serial.println("SIM800C time (NITZ -> NTP fallback),UART4 + PWRKEY PE0(NPN)");

 sim.begin(SIM_BAUD);

 // Включение без риска "выключить",если уже включен
 if (!sim.ensureOnSmart(1500,1200,8000,12000)) {
 Serial.println("Modem not ready (AT/URC). Check wiring/power/baud.");
 return;
 }

 if (!sim.init(12000)) {
 Serial.println("SIM init failed.");
 return;
 }

 Serial.println("SIM ready.");

 tLast = millis() - PERIOD_MS; // первый запрос сразу
}

void loop() {
 if (millis() - tLast >= PERIOD_MS) {
 tLast += PERIOD_MS;

 String resp;
 bool ok = sim.getTimeSmart(resp,APN,USER,PASS,NTP_SERVER,TZ_HOURS,true);
 if (!ok) {
 Serial.print("Get time failed:");
 Serial.println(resp);
 } else {
 printTime(resp);
 }
 }
}
Примечания по NTP на SIM800C
AT+CNTP не возвращает само время в ответе,а обновляет RTC модема,после чего время читается через AT+CCLK?.
Параметр timezone в AT+CNTP="server",<tz> у SIM800 обычно в часах. А в AT+CCLK timezone показывается в четвертях часа (±zz).
Что нужно уточнить/подстроить при необходимости
Ваш APN (и нужны ли USER/PASS).
Реальная скорость UART на модеме (часто 9600,но бывает 115200).
Хотите ли вы держать bearer открытым постоянно (быстрее NTP,но расход/состояние сети) или как сейчас — открывать/закрывать на синхронизацию.
Если скажете оператора/регион и APN — подставлю корректные значения и добавлю более точную диагностику кодов +CNTP: (DNS/таймаут/нет PDP и т.п.).

#include "sim800c_time.h"

SIM800C_Time::SIM800C_Time(HardwareSerial &serial, uint8_t pwrKeyPin)
    : _serial(serial), _pwrKeyPin(pwrKeyPin), _lastRequest(0) {}

void SIM800C_Time::begin(unsigned long baud) {
    pinMode(_pwrKeyPin, OUTPUT);
    digitalWrite(_pwrKeyPin, LOW);
    _serial.begin(baud);
}

void SIM800C_Time::powerOn() {
    // Имитация нажатия кнопки
    digitalWrite(_pwrKeyPin, HIGH);
    delay(1500);
    digitalWrite(_pwrKeyPin, LOW);
    delay(5000); // ждем загрузки модуля
}

void SIM800C_Time::powerOff() {
    digitalWrite(_pwrKeyPin, HIGH);
    delay(1500);
    digitalWrite(_pwrKeyPin, LOW);
    delay(2000);
}

void SIM800C_Time::sendAT(const char* command) {
    _serial.println(command);
}

bool SIM800C_Time::waitForResponse(const char* expected, unsigned long timeout) {
    unsigned long start = millis();
    String response = "";
    while ((millis() - start) < timeout) {
        while (_serial.available()) {
            char c = _serial.read();
            response += c;
            if (response.indexOf(expected) != -1) return true;
        }
    }
    return false;
}

bool SIM800C_Time::getTime(String &outTime) {
    sendAT("AT+CCLK?");
    unsigned long start = millis();
    String response = "";
    while ((millis() - start) < 2000) {
        while (_serial.available()) {
            char c = _serial.read();
            response += c;
            if (response.indexOf("CCLK:") != -1) {
                int idx1 = response.indexOf('"');
                int idx2 = response.indexOf('"', idx1+1);
                if (idx1 != -1 && idx2 != -1) {
                    outTime = response.substring(idx1 + 1, idx2);
                    return true;
                }
            }
        }
    }
    return false;
}

void SIM800C_Time::handle() {
    if (millis() - _lastRequest >= 60000 || _lastRequest == 0) {
        String timeStr;
        if (getTime(timeStr)) {
            Serial.print("Текущее время (SIM800C): ");
            Serial.println(timeStr);
        } else {
            Serial.println("Не удалось получить время от SIM800C");
        }
        _lastRequest = millis();
    }
}

class SIM800TimeClient {
public:
 SIM800TimeClient(HardwareSerial& serial,uint8_t pwrkeyPin,bool pwrkeyActiveHigh = true);

 void begin(uint32_t baud);

void SIM800TimeClient::_pollURC(uint32_t ms) {
 _dbgKV(F("SIM800:wait URC ms="),String(ms));
 String line;
 uint32_t t0 = millis();
 while (millis() - t0 < ms) {
 while (_sim.available()) {
 char c = (char)_sim.read();
 if (c == '\r') continue;

 if (c == '\n') {
 if (line.length()) {
 if (_dbg) {
 _dbg->print('['); _dbg->print(millis()); _dbg->print("] ");
 _dbg->print(F("URC:"));
 _dbg->println(line);
 }
 if (line.indexOf("RDY") >= 0) _seenRDY = true;
 if (line.indexOf("Call Ready") >= 0) _seenCallReady = true;
 if (line.indexOf("SMS Ready") >= 0) _seenSMSReady = true;
 line = "";
 }
 } else {
 if (line.length() < 96) line += c;
 }
 }
 delay(2);
 }
 if (_dbg) {
 _dbgKV(F("URC seen RDY="),String(_seenRDY ? "1" :"0"));
 _dbgKV(F("URC seen CallReady="),String(_seenCallReady ? "1" :"0"));
 _dbgKV(F("URC seen SMSReady="),String(_seenSMSReady ? "1" :"0"));
 }
}

bool SIM800TimeClient::parseCCLK(const String& resp,
 int& year,int& month,int& day,
 int& hour,int& minute,int& second,
 int& tzQuarterHours) {
 int q1 = resp.indexOf('\"');
 int q2 = resp.indexOf('\"',q1 + 1);
 if (q1 < 0 || q2 < 0) return false;

 String v = resp.substring(q1 + 1,q2);
 if (v.length() < 17) return false;

 int yy = v.substring(0,2).toInt();
 month = v.substring(3,5).toInt();
 day = v.substring(6,8).toInt();
 hour = v.substring(9,11).toInt();
 minute= v.substring(12,14).toInt();
 second= v.substring(15,17).toInt();
 year = 2000 + yy;

 tzQuarterHours = 0;
 if (v.length() >= 20) {
 char sign = v.charAt(17);
 int zz = v.substring(18).toInt();
 tzQuarterHours = (sign == '-') ? -zz :zz;
 }
 return true;
}

bool SIM800TimeClient::isCCLKValid(const String& resp,int minYear) {
 int y,mo,d,h,mi,s,tzq;
 if (!parseCCLK(resp,y,mo,d,h,mi,s,tzq)) return false;
 if (y < minYear) return false;
 if (mo < 1 || mo > 12) return false;
 if (d < 1 || d > 31) return false;
 if (h < 0 || h > 23) return false;
 if (mi < 0 || mi > 59) return false;
 if (s < 0 || s > 59) return false;
 return true;
}

bool SIM800TimeClient::_ntpRequest(const char* server,int tzHours,uint32_t timeoutMs) {
 _dbgln(F("SIM800:NTP request"));

 if (!_sendAT("AT+CNTPCID=1","OK",2000)) return false;

 {
 String cmd = String("AT+CNTP=\"") + server + "\"," + String(tzHours);
 if (!_sendAT(cmd.c_str(),"OK",3000)) return false;
 }

 while (_sim.available()) _sim.read();

 if (_dbg && _logAT) _dbgln(F(">> AT+CNTP"));
 _sim.print("AT+CNTP\r");

 String resp = _readAll(timeoutMs);

 if (_dbg && _logAT) {
 _dbgln(F("<<"));
 if (_dbg) _dbg->println(resp);
 }

 int p = resp.indexOf("+CNTP:");
 if (p < 0) return false;

 int colon = resp.indexOf(':',p);
 if (colon < 0) return false;
 int end = resp.indexOf('\n',colon);
 if (end < 0) end = resp.length();

 String tail = resp.substring(colon + 1,end);
 tail.trim();
 int code = tail.toInt();

 if (_dbg) _dbgKV(F("CNTP code="),String(code));

 return (code == 1);
}

Если SIM800C “живёт” ~5 секунд и потом как будто выключается,это почти всегда одно из двух:

Просадка питания VBAT (модем стартует,начинает регистрацию в сети/передачу,даёт импульсные токи 
до 2 А → VBAT падает → модем уходит в reset/PowerDown).
PWRKEY фактически остаётся “нажатым” (через NPN),и модем воспринимает это как команду выключения 
(у SIM800 включение и выключение делаются одним и тем же входом PWRKEY разной длительностью/моментом).
Ниже — как быстро определить причину и что исправить,плюс патч в код,чтобы PWRKEY гарантированно был отпущен.

1) Как понять,что именно происходит (по логам)
Включите ваш подробный лог URC/ответов и посмотрите,что печатает модем перед “отключением”:

Если перед исчезновением связи появляется строка типа NORMAL POWER DOWN — это почти наверняка PWRKEY/команда выключения.
Если появляются UNDER-VOLTAGE WARNING / UNDER-VOLTAGE POWER DOWN или модем циклически печатает 
RDY снова и снова — это питание (просадка VBAT).
Если вообще ничего не успевает вывести,а потом снова RDY — тоже часто питание.
Если у вас нет этих строк,но просто перестаёт отвечать на AT — всё равно вероятнее питание или удержание PWRKEY.

2) Аппаратные причины (питание) — самое частое
Даже при “постоянном питании” модем может выключаться из‑за просадки.

Минимум,что нужно:

VBAT 3.8–4.2 В (не 5 В!),источник с пиковым током 2 А.
Конденсаторы рядом с VBAT модема:
470–1000 µF low ESR (или тантал/полимер) + 100 nF керамика.
Толстые дорожки/провода питания и земли,общий GND с МК.
Как проверить быстро: мультиметр часто не успевает. Нужен осциллограф на VBAT. 
Если видите провалы ниже ~3.4–3.5 В в моменты активности — это причина.

3) PWRKEY через NPN:почему может “выключать” через ~5 сек
При NPN (коллектор на PWRKEY,эмиттер на GND,база от PE0 через резистор):

PE0=HIGH → транзистор открыт → PWRKEY на земле (“нажато”).
Проблемы:

Нет подтяжки базы вниз → при старте/ресете МК база может всплывать,транзистор открыт,PWRKEY “нажат” слишком долго.
PE0 на старте core может кратковременно становиться HIGH/плавающим.
Обязательный элемент: резистор 47–100 кОм с базы на GND (base pulldown),чтобы транзистор точно был закрыт пока МК не инициализировался.

4) Патч в код:сделать PWRKEY “отпущенным” аппаратно и программно
Лучше держать PE0 не как OUTPUT LOW,а как INPUT_PULLDOWN (если доступно) когда “не нажимаем”,и переводить в OUTPUT только на время импульса. Тогда даже при ошибке меньше шанс “удержать кнопку”.

Измените методы управления PWRKEY в SIM800Time.cpp
Замените _setPWRKEYIdle() и _setPWRKEYPressed() на такие (под вашу схему:activeHigh=true,NPN тянет к земле):

cpp Копировать
void SIM800TimeClient::_setPWRKEYIdle() {
 // Отпущено:транзистор точно закрыт
 pinMode(_pwrkeyPin,INPUT_PULLDOWN); // если INPUT_PULLDOWN не поддерживается вашим core -> INPUT
}

void SIM800TimeClient::_setPWRKEYPressed() {
 // Нажато:открыть транзистор
 pinMode(_pwrkeyPin,OUTPUT);
 digitalWrite(_pwrkeyPin,HIGH);
}
Если ваш core не поддерживает INPUT_PULLDOWN,используйте INPUT и обязательно внешний pulldown на базе NPN.

И ещё:очень важно в самом начале setup()
До любых других действий (даже до Serial.begin) жёстко “отпустите” PWRKEY:

cpp Копировать
pinMode(PE0,INPUT_PULLDOWN); // или OUTPUT + LOW,если pulldown недоступен
5) Что сделать прямо сейчас (минимальный план)

Получать время с SIM800C не блокируя общий процесс


Нужно уйти от delay()/циклов ожидания и сделать машину состояний + “тикер”,который 
вы вызываете в loop() очень часто. Тогда общий процесс (ваши задачи) не блокируется,а SIM800C обслуживается “фоном”.

Ниже — готовый вариант в отдельных файлах .h / .cpp + пример .ino для STM32F103 + UART4 (Serial4) + PWRKEY на PE0 через NPN.

Поддерживает:

включение модема (если он не отвечает),
инициализацию,
запрос времени раз в минуту,
fallback на NTP по GPRS,если AT+CCLK? невалиден (NITZ не пришёл).

void SIM800TimeAsync::_serviceState() {
 const uint32_t now = millis();

 // планировщик (только в IDLE)
 if (_st == State::IDLE) {
 if (_forcePoll || (int32_t)(now - _nextPollMs) >= 0) {
 _forcePoll = false;
 _nextPollMs = now + _periodMs;
 _enter(State::WAIT_NETWORK);
 return;
 }
 return;
 }

 switch (_st) {
 case State::STOPPED:
 return;

 case State::PROBE_AT:
 // не блокируем:просто стартуем AT и ждём
 if (!_atPending) _startAT("AT","OK",1500);
 else {
 if (_atDoneOK()) {
 _enter(State::INIT_ATE0);
 } else if (_atDoneTimeout()) {
 _enter(State::PWRKEY_PULSE);
 }
 }
 return;

 case State::PWRKEY_PULSE:
 // короткая “нажатие-отпускание” без delay:делаем по времени
 if (now - _stEnterMs < 50) {
 _pwrkeyIdle();
 } else if (now - _stEnterMs < 1300) {
 _pwrkeyPress();
 } else {
 _pwrkeyRelease();
 _seenRDY = _seenCallReady = _seenSMSReady = false;
 _enter(State::WAIT_URC);
 }
 return;

 case State::WAIT_URC:
 // ждём URC до 8 сек,но даже если не увидим - потом проверим AT
 if (_seenRDY || _seenCallReady || _seenSMSReady || (now - _stEnterMs) > 8000) {
 _enter(State::WAIT_AT_READY);
 }
 return;

 case State::WAIT_AT_READY:
 if (!_atPending) _startAT("AT","OK",12000);
 else {
 if (_atDoneOK()) _enter(State::INIT_ATE0);
 else if (_atDoneTimeout()) _enter(State::STOPPED);
 }
 return;

 case State::INIT_ATE0:
 if (!_atPending) _startAT("ATE0","OK",1500);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::INIT_CMEE);
 return;

 case State::INIT_CMEE:
 if (!_atPending) _startAT("AT+CMEE=2","OK",2000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::INIT_CLTS);
 return;

 case State::INIT_CLTS:
 if (!_atPending) _startAT("AT+CLTS=1","OK",2000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::INIT_SAVE);
 return;

 case State::INIT_SAVE:
 if (!_atPending) _startAT("AT&W","OK",3000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::IDLE);
 return;

 case State::WAIT_NETWORK:
 if (!_atPending) _startAT("AT+CREG?","+CREG:",1500);
 else {
 if (_atDoneTimeout()) {
 // повторим через секунду,не блокируя
 if (now - _stEnterMs > 60000) { // общий таймаут 60 сек
 _enter(State::IDLE);
 } else {
 _atPending = false;
 _atResp = "";
 // подождём 1 сек перед следующим CREG?
 if (now - _stEnterMs > 1000) {
 // просто продолжим,_startAT вызовется в следующем tick()
 }
 }
 } else if (_atMatchedExpect()) {
 // ищем ,1 или ,5
 if (_atResp.indexOf(",1") >= 0 || _atResp.indexOf(",5") >= 0) {
 _enter(State::REQ_CCLK);
 } else {
 // ещё не зарегистрирован
 _atPending = false;
 _atResp = "";
 }
 }
 }
 return;

 case State::REQ_CCLK:
 if (!_atPending) _startAT("AT+CCLK?","+CCLK:",1500);
 else {
 if (_atDoneOK() || _atMatchedExpect()) {
 // сохраним сырой ответ (весь буфер)
 _lastCCLK = _atResp;
 _newTime = true;

 if (!_isCCLKValid(_lastCCLK,2020) && _ntp.enableFallback && _ntp.apn && _ntp.apn[0]) {
 _enter(State::NTP_CGATT);
 } else {
 _enter(State::IDLE);
 }
 } else if (_atDoneTimeout()) {
 _enter(State::IDLE);
 }
 }
 return;

 // -------- NTP fallback (не блокирует) --------
 case State::NTP_CGATT:
 if (!_atPending) _startAT("AT+CGATT?","+CGATT:",2000);
 else {
 if (_atDoneTimeout()) { _enter(State::NTP_SAPBR_CLOSE); return; }
 if (_atMatchedExpect()) {
 if (_atResp.indexOf("+CGATT:1") >= 0) _enter(State::NTP_SAPBR_CFG1);
 else { _atPending = false; _atResp=""; _startAT("AT+CGATT=1","OK",5000); }
 }
 }
 return;

 case State::NTP_SAPBR_CFG1:
 if (!_atPending) _startAT("AT+SAPBR=3,1,\"Contype\",\"GPRS\"","OK",2000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::NTP_SAPBR_CFG_APN);
 return;

 case State::NTP_SAPBR_CFG_APN:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"APN\",\"") + _ntp.apn + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atDoneOK() || _atDoneTimeout()) {
 _enter((_ntp.user && _ntp.user[0]) ? State::NTP_SAPBR_CFG_USER
 :((_ntp.pass && _ntp.pass[0]) ? State::NTP_SAPBR_CFG_PWD :State::NTP_SAPBR_OPEN));
 }
 return;
 }

 case State::NTP_SAPBR_CFG_USER:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"USER\",\"") + _ntp.user + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atDoneOK() || _atDoneTimeout()) {
 _enter((_ntp.pass && _ntp.pass[0]) ? State::NTP_SAPBR_CFG_PWD :State::NTP_SAPBR_OPEN);
 }
 return;
 }

 case State::NTP_SAPBR_CFG_PWD:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"PWD\",\"") + _ntp.pass + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atDoneOK() || _atDoneTimeout()) {
 _enter(State::NTP_SAPBR_OPEN);
 }
 return;
 }

 case State::NTP_SAPBR_OPEN:
 if (!_atPending) _startAT("AT+SAPBR=1,1","OK",45000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::NTP_CNTPCID);
 return;

 case State::NTP_CNTPCID:
 if (!_atPending) _startAT("AT+CNTPCID=1","OK",2000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::NTP_CNTP_SET);
 return;

 case State::NTP_CNTP_SET:{
 if (!_atPending) {
 String cmd = String("AT+CNTP=\"") + (_ntp.server ? _ntp.server :"pool.ntp.org") + "\"," + String(_ntp.tzHours);
 _startAT(cmd.c_str(),"OK",3000);
 } else if (_atDoneOK() || _atDoneTimeout()) {
 _enter(State::NTP_CNTP_START);
 }
 return;
 }

 case State::NTP_CNTP_START:
 // CNTP возвращает +CNTP:<code> асинхронно,ждём его в _handleLine
 if (!_atPending) _startAT("AT+CNTP","+CNTP:",20000);
 else {
 if (_atMatchedExpect()) {
 // успех обычно +CNTP:1
 if (_atResp.indexOf("+CNTP:1") >= 0 || _atResp.indexOf("+CNTP:1") >= 0) {
 _enter(State::NTP_SAPBR_CLOSE);
 } else {
 _enter(State::NTP_SAPBR_CLOSE);
 }
 } else if (_atDoneTimeout()) {
 _enter(State::NTP_SAPBR_CLOSE);
 }
 }
 return;

 case State::NTP_SAPBR_CLOSE:
 if (!_atPending) _startAT("AT+SAPBR=0,1","OK",8000);
 else if (_atDoneOK() || _atDoneTimeout()) _enter(State::REQ_CCLK_AFTER_NTP);
 return;

 case State::REQ_CCLK_AFTER_NTP:
 if (!_atPending) _startAT("AT+CCLK?","+CCLK:",1500);
 else {
 if (_atDoneOK() || _atMatchedExpect()) {
 _lastCCLK = _atResp;
 _newTime = true;
 }
 _enter(State::IDLE);
 }
 return;
 }
}

// ---------------- UART + URC ----------------
void SIM800TimeAsync::_readUartNonBlocking() {
 while (_sim.available()) {
 char c = (char)_sim.read();
 if (c == '\r') continue;

 if (c == '\n') {
 if (_lineBuf.length()) {
 _handleLine(_lineBuf);
 _lineBuf = "";
 }
 } else {
 if (_lineBuf.length() < 160) _lineBuf += c;
 }
 }
}

void SIM800TimeAsync::_handleLine(const String& line) {
 // URC
 if (line.indexOf("RDY") >= 0) _seenRDY = true;
 if (line.indexOf("Call Ready") >= 0) _seenCallReady = true;
 if (line.indexOf("SMS Ready") >= 0) _seenSMSReady = true;

 // Лог входящих строк
 if (_dbg && _logAT) {
 _dbg->print('['); _dbg->print(millis()); _dbg->print("] ");
 _dbg->print(F("<< "));
 _dbg->println(line);
 }

 // Если идет AT-транзакция — копим ответ
 if (_atPending) {
 _atResp += line;
 _atResp += "\n";

 // быстрый финиш по OK/ERROR/expect — окончательно обработаем в _serviceState()
 // (здесь лишь накапливаем)
 }
}

// ---------------- AT helpers ----------------
void SIM800TimeAsync::_startAT(const char* cmd,const char* expect,uint32_t timeoutMs) {
 // очистить "хвост" от предыдущего
 _atResp = "";
 _atExpect = expect;
 _atPending = true;
 _atDeadlineMs = millis() + timeoutMs;

 if (_dbg && _logAT) {
 _dbg->print('['); _dbg->print(millis()); _dbg->print("] ");
 _dbg->print(F(">> "));
 _dbg->println(cmd);
 }

 _sim.print(cmd);
 _sim.print("\r");
}

bool SIM800TimeAsync::_atDoneOK() const {
 if (!_atPending) return true;
 if ((int32_t)(millis() - _atDeadlineMs) >= 0) return false;
 return (_atResp.indexOf("\nOK") >= 0) || (_atResp.startsWith("OK")) || (_atResp.indexOf("OK\n") >= 0);
}

bool SIM800TimeAsync::_atDoneTimeout() const {
 if (!_atPending) return false;
 return (int32_t)(millis() - _atDeadlineMs) >= 0;
}

bool SIM800TimeAsync::_atMatchedExpect() const {
 if (!_atPending || !_atExpect) return false;
 return _atResp.indexOf(_atExpect) >= 0;
}

// ---------------- PWRKEY helpers ----------------
// Для вашей схемы (NPN тянет к GND):лучше НЕ держать кнопку.

 // отпущено
#if defined(INPUT_PULLDOWN)
 pinMode(_pwrkeyPin,INPUT_PULLDOWN);
#else
 pinMode(_pwrkeyPin,OUTPUT);
 digitalWrite(_pwrkeyPin,_pwrkeyActiveHigh ? LOW :HIGH);
#endif
}

void SIM800TimeAsync::_pwrkeyPress() {
 pinMode(_pwrkeyPin,OUTPUT);
 digitalWrite(_pwrkeyPin,_pwrkeyActiveHigh ? HIGH :LOW);
}

void SIM800TimeAsync::_pwrkeyRelease() {
 _pwrkeyIdle();
}

// ---------------- time parse/valid ----------------
bool SIM800TimeAsync::parseCCLK(const String& resp,
 int& year,int& month,int& day,
 int& hour,int& minute,int& second,
 int& tzQuarterHours) {
 int q1 = resp.indexOf('\"');
 int q2 = resp.indexOf('\"',q1 + 1);
 if (q1 < 0 || q2 < 0) return false;

 String v = resp.substring(q1 + 1,q2); // yy/MM/dd,hh:mm:ss±zz
 if (v.length() < 17) return false;

 int yy = v.substring(0,2).toInt();
 month = v.substring(3,5).toInt();
 day = v.substring(6,8).toInt();
 hour = v.substring(9,11).toInt();
 minute = v.substring(12,14).toInt();
 second = v.substring(15,17).toInt();
 year = 2000 + yy;

 tzQuarterHours = 0;
 if (v.length() >= 20) {
 char sign = v.charAt(17);
 int zz = v.substring(18).toInt();
 tzQuarterHours = (sign == '-') ? -zz :zz;
 }
 return true;
}

bool SIM800TimeAsync::_isCCLKValid(const String& resp,int minYear) {
 int y,mo,d,h,mi,s,tzq;
 if (!parseCCLK(resp,y,mo,d,h,mi,s,tzq)) return false;
 if (y < minYear) return false;
 if (mo < 1 || mo > 12) return false;
 if (d < 1 || d > 31) return false;
 if (h < 0 || h > 23) return false;
 if (mi < 0 || mi > 59) return false;
 if (s < 0 || s > 59) return false;
 return true;
}

 В вашем RealtimeClock::setTime(...) 4‑й параметр dayOfWeek по факту должен быть 1..7,потому что:

в getTime() вы делаете constrain(t.dayOfWeek,1,7);
getDayOfWeekStr() ожидает 1..7 и мапит 1=Mon … 7=Sun

А ваша фраза “0‑воскресенье..6‑суббота” — это формат tm_wday. Поэтому правильно сделать так:

считаем 0..6 (0=Sun),
затем конвертируем в формат DS3231:Sunday(0) → 7,остальные оставляем как есть.
Ниже — готовый код,который:

// Конверсия под вашу RTC/DS3231:1=Mon..6=Sat,7=Sun
static uint8_t ds3231DowFromDow0(uint8_t dow0) {
 return (dow0 == 0) ? 7 :dow0; // Sun->7,Mon..Sat->1..6

// ---------- UTC conversion helpers ----------
static int32_t daysFromCivil(int y,int m,int d) {
 y -= (m <= 2);
 const int era = (y >= 0 ? y :y - 399) / 400;
 const unsigned yoe = (unsigned)(y - era * 400);
 const unsigned doy = (153 * (m + (m > 2 ? -3 :9)) + 2) / 5 + (unsigned)d - 1;
 const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
 return (int32_t)(era * 146097 + (int)doe - 719468); // to 1970-01-01
}

static void civilFromDays(int32_t z,int &y,int &m,int &d) {
 z += 719468;
 const int era = (z >= 0 ? z :z - 146096) / 146097;
 const unsigned doe = (unsigned)(z - era * 146097);
 const unsigned yoe = (doe - doe/1460 + doe/36524 - doe/146096) / 365;
 y = (int)yoe + era * 400;
 const unsigned doy = doe - (365*yoe + yoe/4 - yoe/100);
 const unsigned mp = (5*doy + 2) / 153;
 d = (int)(doy - (153*mp + 2)/5 + 1);
 m = (int)(mp + (mp < 10 ? 3 :-9));
 y += (m <= 2);
}

static int64_t toEpochSeconds(int y,int mo,int d,int h,int mi,int s) {
 int32_t days = daysFromCivil(y,mo,d);
 return (int64_t)days * 86400LL + (int64_t)h * 3600LL + (int64_t)mi * 60LL + (int64_t)s;
}

static void fromEpochSeconds(int64_t t,int &y,int &mo,int &d,int &h,int &mi,int &s) {
 int64_t days = t / 86400LL;
 int64_t rem = t % 86400LL;
 if (rem < 0) { rem += 86400LL; days -= 1; }

 civilFromDays((int32_t)days,y,mo,d);

 h = (int)(rem / 3600LL);
 rem %= 3600LL;
 mi = (int)(rem / 60LL);
 s = (int)(rem % 60LL);
}

// Local(+tzq) -> UTC. tzq = четверти часа,local = utc + tz
static void localToUTC(int &y,int &mo,int &d,int &h,int &mi,int &s,int tzQuarterHours) {
 const int offsetMin = tzQuarterHours * 15;
 int64_t epochLocal = toEpochSeconds(y,mo,d,h,mi,s);
 int64_t epochUTC = epochLocal - (int64_t)offsetMin * 60LL;
 fromEpochSeconds(epochUTC,y,mo,d,h,mi,s);
}

void SIM800TimeAsync::_serviceState() {
 const uint32_t now = millis();

 // планировщик (только в IDLE)
 if (_st == State::IDLE) {
 if (_forcePoll || (int32_t)(now - _nextPollMs) >= 0) {
 _forcePoll = false;
 _nextPollMs = now + _periodMs;
 _enter(State::WAIT_NETWORK);
 }
 return;
 }

 switch (_st) {
 case State::STOPPED:
 return;

 case State::PROBE_AT:
 if (!_atPending) {
 _startAT("AT","OK",1500);
 } else {
 if (_atOk) {
 _enter(State::INIT_ATE0);
 } else if (_atDoneTimeout()) {
 _enter(State::PWRKEY_PULSE);
 }
 }
 return;

 case State::PWRKEY_PULSE:
 // “нажатие” без delay:50мс idle,затем ~1.25с press
 if (now - _stEnterMs < 50) {
 _pwrkeyIdle();
 } else if (now - _stEnterMs < 1300) {
 _pwrkeyPress();
 } else {
 _pwrkeyRelease();
 _seenRDY = _seenCallReady = _seenSMSReady = false;
 _enter(State::WAIT_URC);
 }
 return;

 case State::WAIT_URC:
 if (_seenRDY || _seenCallReady || _seenSMSReady || (now - _stEnterMs) > 8000) {
 _enter(State::WAIT_AT_READY);
 }
 return;

 case State::WAIT_AT_READY:
 if (!_atPending) {
 _startAT("AT","OK",12000);
 } else {
 if (_atOk) {
 _enter(State::INIT_ATE0);
 } else if (_atDoneTimeout()) {
 _enter(State::STOPPED);
 }
 }
 return;

 case State::INIT_ATE0:
 if (!_atPending) _startAT("ATE0","OK",1500);
 else if (_atOk || _atDoneTimeout()) _enter(State::INIT_CMEE);
 return;

 case State::INIT_CMEE:
 if (!_atPending) _startAT("AT+CMEE=2","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::INIT_CLTS);
 return;

 case State::INIT_CLTS:
 if (!_atPending) _startAT("AT+CLTS=1","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::INIT_SAVE);
 return;

 case State::INIT_SAVE:
 if (!_atPending) _startAT("AT&W","OK",3000);
 else if (_atOk || _atDoneTimeout()) _enter(State::IDLE);
 return;

 case State::WAIT_NETWORK:
 if (!_atPending) {
 _startAT("AT+CREG?","+CREG:",1500);
 } else {
 if (_atDoneTimeout()) {
 // общий таймаут ожидания сети:60с
 if (now - _stEnterMs > 60000) {
 _enter(State::IDLE);
 } else {
 // повторим без блокировки
 _atPending = false;
 _atResp = "";
 _atOk = _atErr = false;
 }
 } else if (_atMatchedExpect()) {
 if (_atResp.indexOf(",1") >= 0 || _atResp.indexOf(",5") >= 0) {
 _enter(State::REQ_CCLK);
 } else {
 // ещё не зарегистрирован — повторим
 _atPending = false;
 _atResp = "";
 _atOk = _atErr = false;
 }
 }
 }
 return;

 case State::REQ_CCLK:
 if (!_atPending) {
 _startAT("AT+CCLK?","+CCLK:",2000);
 } else {
 // ВАЖНО:для CCLK считаем успешным только наличие +CCLK:
 if (_atMatchedExpect()) {
 // _lastCCLK установится в _handleLine() когда придет строка +CCLK:
 // но на всякий случай:
 if (_lastCCLK.length() == 0) _lastCCLK = _atResp;

 // событие времени поднимем только если реально есть +CCLK:
 if (_lastCCLK.indexOf("+CCLK:") >= 0) _newTime = true;

 if (!_isCCLKValid(_lastCCLK,2020) && _ntp.enableFallback && _ntp.apn && _ntp.apn[0]) {
 _enter(State::NTP_CGATT);
 } else {
 _enter(State::IDLE);
 }
 } else if (_atDoneTimeout()) {
 _enter(State::IDLE);
 }
 }
 return;

 // -------- NTP fallback (не блокирует) --------
 case State::NTP_CGATT:
 if (!_atPending) _startAT("AT+CGATT?","+CGATT:",2000);
 else {
 if (_atDoneTimeout()) {
 _enter(State::NTP_SAPBR_CLOSE);
 return;
 }
 if (_atMatchedExpect()) {
 if (_atResp.indexOf("+CGATT:1") >= 0) {
 _enter(State::NTP_SAPBR_CFG1);
 } else {
 // попробуем приаттачиться
 _atPending = false;
 _startAT("AT+CGATT=1","OK",5000);
 }
 }
 }
 return;

 case State::NTP_SAPBR_CFG1:
 if (!_atPending) _startAT("AT+SAPBR=3,1,\"Contype\",\"GPRS\"","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::NTP_SAPBR_CFG_APN);
 return;

 case State::NTP_SAPBR_CFG_APN:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"APN\",\"") + _ntp.apn + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter((_ntp.user && _ntp.user[0]) ? State::NTP_SAPBR_CFG_USER
 :((_ntp.pass && _ntp.pass[0]) ? State::NTP_SAPBR_CFG_PWD :State::NTP_SAPBR_OPEN));
 }
 return;
 }

 case State::NTP_SAPBR_CFG_USER:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"USER\",\"") + _ntp.user + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter((_ntp.pass && _ntp.pass[0]) ? State::NTP_SAPBR_CFG_PWD :State::NTP_SAPBR_OPEN);
 }
 return;
 }

 case State::NTP_SAPBR_CFG_PWD:{
 if (!_atPending) {
 String cmd = String("AT+SAPBR=3,1,\"PWD\",\"") + _ntp.pass + "\"";
 _startAT(cmd.c_str(),"OK",2000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter(State::NTP_SAPBR_OPEN);
 }
 return;
 }

 case State::NTP_SAPBR_OPEN:
 if (!_atPending) _startAT("AT+SAPBR=1,1","OK",45000);
 else if (_atOk || _atDoneTimeout()) _enter(State::NTP_CNTPCID);
 return;

 case State::NTP_CNTPCID:
 if (!_atPending) _startAT("AT+CNTPCID=1","OK",2000);
 else if (_atOk || _atDoneTimeout()) _enter(State::NTP_CNTP_SET);
 return;

 case State::NTP_CNTP_SET:{
 if (!_atPending) {
 String cmd = String("AT+CNTP=\"") + (_ntp.server ? _ntp.server :"pool.ntp.org") + "\"," + String(_ntp.tzHours);
 _startAT(cmd.c_str(),"OK",3000);
 } else if (_atOk || _atDoneTimeout()) {
 _enter(State::NTP_CNTP_START);
 }
 return;
 }

 case State::NTP_CNTP_START:
 // ждём +CNTP:<code>
 if (!_atPending) _startAT("AT+CNTP","+CNTP:",20000);

 return;

 case State::REQ_CCLK_AFTER_NTP:

 _startAT("AT+CCLK?","+CCLK:",2000);
 } else {
 if (_atMatchedExpect()) {
 if (_lastCCLK.indexOf("+CCLK:") >= 0) _newTime = true;
 }
 _enter(State::IDLE);
 }
 return;
 }

// ---------------- UART + URC ----------------
void SIM800TimeAsync::_readUartNonBlocking() {
 while (_sim.available()) {
 char c = (char)_sim.read();
 if (c == '\r') continue;

 if (c == '\n') {
 if (_lineBuf.length()) {
 _handleLine(_lineBuf);
 _lineBuf = "";
 }
 } else {
 if (_lineBuf.length() < 160) _lineBuf += c;
 }
 }
}

void SIM800TimeAsync::_handleLine(const String& line) {
 // URC признаки
 if (line.indexOf("RDY") >= 0) _seenRDY = true;
 if (line.indexOf("Call Ready") >= 0) _seenCallReady = true;
 if (line.indexOf("SMS Ready") >= 0) _seenSMSReady = true;

 // Лог входящих строк
 if (_dbg && _logAT) {
 _dbg->print('['); _dbg->print(millis()); _dbg->print("] ");
 _dbg->print(F("<< "));
 _dbg->println(line);
 }

 // AT-транзакция:копим ВСЕ строки,чтобы OK/ERROR не терялись
 if (_atPending) {
 if (line == "OK") _atOk = true;
 else if (line == "ERROR") _atErr = true;

 if (line.startsWith("+CCLK:")) {
 // сохраняем "чистую" строку времени
 _lastCCLK = line;
 // событие не поднимаем здесь,поднимаем в state (когда увидели +CCLK:)
 }

 _atResp += line;
 _atResp += "\n";
 }
}

// ---------------- time parse/valid ----------------
bool SIM800TimeAsync::parseCCLK(const String& resp,
 int& year,int& month,int& day,
 int& hour,int& minute,int& second,
 int& tzQuarterHours) {

 int q1 = resp.indexOf('\"');
 int q2 = resp.indexOf('\"',q1 + 1);
 if (q1 < 0 || q2 < 0) return false;

 String v = resp.substring(q1 + 1,q2); // yy/MM/dd,hh:mm:ss±zz
 if (v.length() < 17) return false;

 int yy = v.substring(0,2).toInt();
 month = v.substring(3,5).toInt();
 day = v.substring(6,8).toInt();
 hour = v.substring(9,11).toInt();
 minute = v.substring(12,14).toInt();
 second = v.substring(15,17).toInt();
 year = 2000 + yy;

 tzQuarterHours = 0;
 if (v.length() >= 20) {
 char sign = v.charAt(17);
 int zz = v.substring(18).toInt(); // четверти часа
 tzQuarterHours = (sign == '-') ? -zz :zz;
 }
 return true;
}

bool SIM800TimeAsync::_isCCLKValid(const String& resp,int minYear) {
 int y,mo,d,h,mi,s,tzq;
 if (!parseCCLK(resp,y,mo,d,h,mi,s,tzq)) return false;
 if (y < minYear) return false;
 if (mo < 1 || mo > 12) return false;
 if (d < 1 || d > 31) return false;
 if (h < 0 || h > 23) return false;
 if (mi < 0 || mi > 59) return false;
 if (s < 0 || s > 59) return false;
 return true;
}
